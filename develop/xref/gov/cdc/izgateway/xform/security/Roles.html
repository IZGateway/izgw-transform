1   package gov.cdc.izgateway.xform.security;
2   
3   import java.util.Arrays;
4   import java.util.Collections;
5   import java.util.List;
6   
7   /**
8    * Names of the roles that control access to the endpoints.
9    */
10  public class Roles {
11      // Supported roles
12      /** Role that has full access to all endpoints and organizations. */
13      public static final String ADMIN = "admin";
14      /** Role for systems that send messages to the transformation endpoints. */
15      public static final String XFORM_SENDING_SYSTEM = "xform-sender";
16  
17      /** Role that allows reading pipelines through the API. */
18      public static final String PIPELINE_READER = "pipeline-reader";
19      /** Role that allows creating and updating pipelines through the API. */
20      public static final String PIPELINE_WRITER = "pipeline-writer";
21      /** Role that allows deleting pipelines through the API. */
22      public static final String PIPELINE_DELETER = "pipeline-deleter";
23  
24      /** Role that allows reading organizations through the API. */
25      public static final String ORGANIZATION_READER = "organization-reader";
26      /** Role that allows creating and updating organizations through the API. */
27      public static final String ORGANIZATION_WRITER = "organization-writer";
28      /** Role that allows deleting organizations through the API. */
29      public static final String ORGANIZATION_DELETER = "organization-deleter";
30  
31      /** Role that allows reading solutions through the API. */
32      public static final String SOLUTION_READER = "solution-reader";
33      /** Role that allows creating and updating solutions through the API. */
34      public static final String SOLUTION_WRITER = "solution-writer";
35      /** Role that allows deleting solutions through the API. */
36      public static final String SOLUTION_DELETER = "solution-deleter";
37  
38      /** Header to indicate that request from localhost should not be treated as an admin */
39      public static final String NOT_ADMIN_HEADER = "X-Not-Admin";
40  
41      /** Header to indicate that the request should remove admin role from the principal to test non-admin access.
42       *  This is different from NOT_ADMIN_HEADER in that it is used to remove the admin role regardless of whether
43       *  the request is from localhost or not.
44       */
45      public static final String REMOVE_ADMIN_ROLE_HEADER = "X-Remove-Admin-Role";
46  
47      /** Special role for public access to an API endpoint. Health check endpoints use this role. */
48      public static final String PUBLIC_ACCESS = "public-access";
49  
50      /**
51       * A list of all roles to enable validation of role names
52       * in model elements.
53       */
54      public static final List<String> ALL_ROLES = Collections.unmodifiableList(
55      	Arrays.asList(
56      		ADMIN, XFORM_SENDING_SYSTEM, 
57      		PIPELINE_READER, PIPELINE_WRITER, PIPELINE_DELETER,
58      		ORGANIZATION_READER, ORGANIZATION_WRITER, ORGANIZATION_DELETER,
59      		SOLUTION_READER, SOLUTION_WRITER, SOLUTION_DELETER, PUBLIC_ACCESS
60      	)
61      );
62      private Roles() {}
63  }
