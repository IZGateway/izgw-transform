4   import ca.uhn.hl7v2.model.Message;
5   import gov.cdc.izgateway.xform.context.ServiceContext;
6   import gov.cdc.izgateway.xform.enums.DataFlowDirection;
7   import gov.cdc.izgateway.xform.logging.advice.*;

9   import gov.cdc.izgateway.xform.preconditions.Precondition;
10  import org.aspectj.lang.ProceedingJoinPoint;
11  import org.aspectj.lang.annotation.Around;
12  import org.aspectj.lang.annotation.Aspect;
13  
14  import java.util.UUID;

16  /**
17   * AspectJ aspect that records pipeline, solution, operation and precondition advice
18   * before and after each method annotated with {@code @CaptureXformAdvice}.
19   */
20  @Aspect
21  public class XformAdviceAspect {
22  
23      /**
24       * Creates the aspect. AspectJ creates the instance.
25       */
26      public XformAdviceAspect() {
27          // No initialization is necessary.
28      }
29  
30      /**
31       * Records advice before and after the advised method and marks the advice as errored if the method throws.
32       *
33       * @param joinPoint the advised method call
34       * @return the value returned by the advised method
35       * @throws Throwable the exception thrown by the advised method
36       */
37      @Around("execution(* *(..)) && @annotation(gov.cdc.izgateway.xform.annotations.CaptureXformAdvice)")
38      public Object processXformAdviceAspect(ProceedingJoinPoint joinPoint) throws Throwable {
39          boolean hasErrored = false;
40  
41          addAdvice(joinPoint, MethodDisposition.PREEXECUTION, hasErrored);
42  
43          try {
44              return joinPoint.proceed();
45          } catch (Exception e) {
46              hasErrored = true;
47              throw e;
48          } finally {
49              addAdvice(joinPoint, MethodDisposition.POSTEXECUTION, hasErrored);
50          }
51  
52      }
53  
54      private void addAdvice(ProceedingJoinPoint joinPoint, MethodDisposition methodDisposition, boolean hasErrored) throws HL7Exception {
55          for (Object arg : joinPoint.getArgs()) {
56              if (arg instanceof ServiceContext context) {
57                  XformAdviceCollector.getTransactionData().addAdvice(getXformAdvice(joinPoint, context, methodDisposition, hasErrored));
58                  break;
59              }
60          }
61      }
62  
63      private XformAdviceDTO getXformAdvice(ProceedingJoinPoint joinPoint, ServiceContext context, MethodDisposition methodDisposition, boolean hasErrored) throws HL7Exception {
64          XformAdviceDTO xformAdvice;
65  
66          if (AdviceUtil.isPipelineAdvice(joinPoint.getTarget().getClass().getSimpleName())) {
67              xformAdvice = new PipelineAdviceDTO();
68          } else if (AdviceUtil.isSolutionAdvice(joinPoint.getTarget().getClass().getSimpleName())) {
69              xformAdvice = new SolutionAdviceDTO();
70          } else if (joinPoint.getTarget() instanceof Operation) {
71              xformAdvice = new OperationAdviceDTO();
72          } else if (joinPoint.getTarget() instanceof Precondition) {
73              xformAdvice = new PreconditionAdviceDTO();
74          } else {
75              return null;
76          }
77  
78          populateAdvice(xformAdvice, joinPoint, context, methodDisposition, hasErrored);
79  
80          return xformAdvice;
81      }
82  
83      private void populateAdvice(XformAdviceDTO xformAdvice, ProceedingJoinPoint joinPoint, ServiceContext context, MethodDisposition methodDisposition, boolean hasErrored) throws HL7Exception {
84          String name = "Unknown";
85          UUID id = null;
86          boolean hasTransformed = false;
87  
88          Object targetObject = joinPoint.getTarget();
89          if (targetObject instanceof Advisable advisable) {
90              name = advisable.getName();
91              id = advisable.getId();
92          }
93  
94          if (methodDisposition == MethodDisposition.POSTEXECUTION && xformAdvice instanceof PipelineAdviceDTO) {
95              hasTransformed = context.getCurrentDirection() == DataFlowDirection.REQUEST
96                      ? XformAdviceCollector.getTransactionData().getPipelineAdvice().isRequestTransformed()
97                      : XformAdviceCollector.getTransactionData().getPipelineAdvice().isResponseTransformed();
98          } else if (targetObject instanceof Transformable transformable) {
99              hasTransformed = transformable.hasTransformed();
100         }
101 
102         xformAdvice.setClassName(joinPoint.getTarget().getClass().getSimpleName());
103         xformAdvice.setName(name);
104         xformAdvice.setId(id);
105         xformAdvice.setProcessError(hasErrored);
106         xformAdvice.setOrganizationId(context.getOrganizationId());
107 
108         if ( context.getCurrentDirection() == DataFlowDirection.REQUEST ) {
109             updateRequestMessage(context, methodDisposition, hasTransformed, xformAdvice);
110         } else {
111             updateResponseMessage(context, methodDisposition, hasTransformed, xformAdvice);
112         }
113     }
114 
115     private void updateRequestMessage(ServiceContext context, MethodDisposition methodDisposition, boolean hasTransformed, XformAdviceDTO advice) throws HL7Exception {
116         if ( advice instanceof PipelineAdviceDTO && methodDisposition == MethodDisposition.PREEXECUTION) {
117             advice.setRequest(context.getRequestMessage().encode());
118         } else if ( methodDisposition == MethodDisposition.POSTEXECUTION && hasTransformed) {
119             advice.setTransformedRequest(context.getRequestMessage().encode());
120         }
121     }
122 
123     private void updateResponseMessage(ServiceContext context, MethodDisposition methodDisposition, boolean hasTransformed, XformAdviceDTO advice) throws HL7Exception {
124         Message responseMessage = context.getResponseMessage();
125 
126         if ( advice instanceof PipelineAdviceDTO && methodDisposition == MethodDisposition.PREEXECUTION ) {
127             advice.setResponse(responseMessage == null ? null : responseMessage.encode());
128         } else if ( methodDisposition == MethodDisposition.POSTEXECUTION && hasTransformed) {
129             advice.setTransformedResponse(responseMessage == null ? null : responseMessage.encode());
130         }
131     }
