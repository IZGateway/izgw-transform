1   package gov.cdc.izgateway.xform.logging.advice;

3   import com.fasterxml.jackson.annotation.JsonIgnore;
4   
5   import java.util.UUID;
6   
7   /**
8    * The Advisable interface defines the contract for objects to be logged
9    * within the IZG Xform Service.
10   *
11   * <p>Implementing classes need to provide methods for retrieving a name and a
12   * unique identifier for the advisable object.
13   *
14   * <p>Please <em>note</em> that the Name is set to ignore because, at this time because
15   * of how the available Operation and Precondition lists via the API are
16   * dynamically generated via reflection.  Without this the name was coming
17   * out in API calls when it is not a field that exists in the configuration
18   * JSON itself.
19   *
20   * @since 0.2.0
21   */
22  public interface Advisable {
23      /**
24       * Returns the name of the advisable object that is recorded in the advice.
25       *
26       * @return the name
27       */
28      @JsonIgnore
29      String getName();
30  
31      /**
32       * Returns the unique identifier of the advisable object that is recorded in the advice.
33       *
34       * @return the identifier
35       */
36      UUID getId();
37  }
