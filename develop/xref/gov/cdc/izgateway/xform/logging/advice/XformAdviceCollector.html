3   import gov.cdc.izgateway.logging.RequestContext;
4   import gov.cdc.izgateway.xform.logging.XformTransactionData;

6   /**
7    * Holds the {@link XformTransactionData} of the current thread, to which the advice aspect adds advice.
8    */
9   public class XformAdviceCollector {
10  	private XformAdviceCollector() {}
11  	private static final ThreadLocal<XformTransactionData> transactionDataVar = new ThreadLocal<>();
12  
13      // TODO - See below simplify by casting to XformTransactionData from RequestContext.getTransactionData()
14      // when I set this with the dervied, on read, it should still be the derived.
15  	/**
16  	 * Returns the transaction data of the current thread.
17  	 *
18  	 * @return the transaction data, or null if none is set
19  	 */
20  	public static XformTransactionData getTransactionData() {
21  		return transactionDataVar.get();
22  	}
23  
24  	/**
25  	 * Sets the transaction data of the current thread and of the {@link RequestContext}. A null value clears both.
26  	 *
27  	 * @param transactionData the transaction data, or null to clear
28  	 */
29  	public static void setTransactionData(XformTransactionData transactionData) {
30          RequestContext.setTransactionData(transactionData);
31  		if (transactionData == null) {
32  			clear();
33  			return;
34  		}
35  		transactionDataVar.set(transactionData);
36  	}
37  
38  	/**
39  	 * Removes the transaction data of the current thread and clears the {@link RequestContext}.
40  	 */
41  	public static void clear() {
42  		transactionDataVar.remove();
43          RequestContext.clear();
44  	}
45  }
