1   package gov.cdc.izgateway.xform.logging.advice;
2   
3   import lombok.Data;
4   import lombok.EqualsAndHashCode;
5   
6   import java.util.ArrayList;
7   import java.util.UUID;
8   
9   /**
10   * Collected advice for one pipeline run, with the advice of its solutions and pipeline-level preconditions.
11   */
12  @Data
13  @EqualsAndHashCode(callSuper = false)
14  public class PipelineAdvice extends PipelineAdviceDTO {
15      private ArrayList<SolutionAdvice> solutionAdviceList = new ArrayList<>();
16      private ArrayList<PreconditionAdviceDTO> preconditionAdviceList = new ArrayList<>();
17      private UUID id;
18      private boolean requestTransformed = false;
19      private boolean responseTransformed = false;
20  
21      /**
22       * Creates the collected pipeline advice from the first advice recorded for the pipeline.
23       *
24       * @param advice the pipeline advice to copy
25       */
26      public PipelineAdvice(PipelineAdviceDTO advice) {
27          this.setClassName(advice.getClassName());
28          this.setName(advice.getName());
29          this.setRequest(advice.getRequest());
30          this.setTransformedRequest(advice.getTransformedRequest());
31          this.setResponse(advice.getResponse());
32          this.setTransformedResponse(advice.getTransformedResponse());
33          this.setId(advice.getId());
34      }
35  
36      /**
37       * Returns the collected advice for the solution with the identifier of the given advice, and adds it if it is not in the list.
38       *
39       * @param advice the solution advice to find
40       * @return the collected solution advice
41       */
42      public SolutionAdvice getSolutionAdvice(SolutionAdviceDTO advice) {
43          SolutionAdvice solutionAdvice = new SolutionAdvice(advice.getId(), advice.getClassName(), advice.getName());
44          int adviceIndex = solutionAdviceList.indexOf(solutionAdvice);
45          if ( adviceIndex >= 0 ) {
46              return solutionAdviceList.get(adviceIndex);
47          }
48  
49          solutionAdviceList.add(solutionAdvice);
50  
51          return solutionAdvice;
52      }
53  
54      /**
55       * Adds the pipeline-level precondition advice if it is not in the list.
56       *
57       * @param advice the precondition advice to add
58       */
59      public void addPreconditionAdvice(PreconditionAdviceDTO advice) {
60          int adviceIndex = preconditionAdviceList.indexOf(advice);
61          if ( adviceIndex < 0 ) {
62              preconditionAdviceList.add(advice);
63          }
64      }
65  
66      /**
67       * Returns true if the precondition advice is in the pipeline-level list.
68       *
69       * @param precondition the precondition advice to find
70       * @return true if the list contains advice with the same identifier
71       */
72      public boolean preconditionExists(PreconditionAdviceDTO precondition) {
73          return (preconditionAdviceList.contains(precondition));
74      }
75  }
