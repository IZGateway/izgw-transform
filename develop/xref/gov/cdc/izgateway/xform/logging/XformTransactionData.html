13  /**
14   * Transaction data of the Xform Service, with the advice collected for the pipeline that ran.

17  @Data
18  @EqualsAndHashCode(callSuper = false)
19  @MarkerObjectFieldName("transactionData")
20  @JsonInclude(JsonInclude.Include.ALWAYS)
21  public class XformTransactionData extends TransactionData {
22      private static final String XFORM_TRANSACTION_DATA = "XformTransactionData";

24      // TODO @JsonIgnore
25      private PipelineAdvice pipelineAdvice = null;

27      @JsonIgnore
28      private SolutionAdvice currentSolutionAdvice = null;

46      /**
47       * Adds advice to the collected advice for its type. Null advice is ignored.
48       *
49       * @param advice the advice to add
50       */
51      public void addAdvice(XformAdviceDTO advice) {
52          if ( advice == null )
53              return;
54  
55          if ( advice instanceof PipelineAdviceDTO pAdvice)
56              addAdvice(pAdvice);
57          else if ( advice instanceof SolutionAdviceDTO sAdvice)
58              addAdvice(sAdvice);
59          else if ( advice instanceof OperationAdviceDTO oAdvice)
60              addAdvice(oAdvice);
61          else if ( advice instanceof PreconditionAdviceDTO preAdvice )
62              addAdvice(preAdvice);
63      }
64  
65      /**
66       * Creates the pipeline advice if necessary and updates it from the given advice.
67       *
68       * @param advice the pipeline advice
69       */
70      public void addAdvice(PipelineAdviceDTO advice) {
71  
72          if ( pipelineAdvice == null ) {
73              pipelineAdvice = new PipelineAdvice(advice);
74          }
75  
76          pipelineAdvice.setName(advice.getName());
77          pipelineAdvice.setId(advice.getId());
78          pipelineAdvice.setProcessError(advice.isProcessError());
79          pipelineAdvice.setOrganizationId(advice.getOrganizationId());
80          if ( advice.getTransformedRequest() != null)
81              pipelineAdvice.setTransformedRequest(advice.getTransformedRequest());
82          if ( advice.getResponse() != null )
83              pipelineAdvice.setResponse(advice.getResponse());
84          if ( advice.getTransformedResponse() != null )
85              pipelineAdvice.setTransformedResponse(advice.getTransformedResponse());
86      }
87  
88      /**
89       * Makes the solution the current solution and records its errors and transformed messages.
90       *
91       * @param advice the solution advice
92       */
93      public void addAdvice(SolutionAdviceDTO advice) {
94  
95          currentSolutionAdvice = pipelineAdvice.getSolutionAdvice(advice);
96          currentSolutionAdvice.setOrganizationId(advice.getOrganizationId());
97  
98          if ( advice.isProcessError() ) {
99              currentSolutionAdvice.setProcessError(true);
100         }
101 
102         if ( advice.getTransformedRequest() != null ) {
103             currentSolutionAdvice.setTransformedRequest(advice.getTransformedRequest());
104             pipelineAdvice.setRequestTransformed(true);
105         }
106         if ( advice.getTransformedResponse() != null ) {
107             currentSolutionAdvice.setTransformedResponse(advice.getTransformedResponse());
108             pipelineAdvice.setResponseTransformed(true);
109         }
110     }
111 
112     /**
113      * Adds operation advice to the current solution if the operation transformed the request or the response.
114      *
115      * @param advice the operation advice
116      */
117     public void addAdvice(OperationAdviceDTO advice) {
118         if (advice.getTransformedRequest() != null)
119             currentSolutionAdvice.addRequestOperationAdvice(advice);
120         if (advice.getTransformedResponse() != null)
121             currentSolutionAdvice.addResponseOperationAdvice(advice);
122 
123     }
124 
125     /**
126      * Adds precondition advice to the pipeline if no solution has run, otherwise to the current solution.
127      *
128      * @param advice the precondition advice
129      */
130     public void addAdvice(PreconditionAdviceDTO advice) {
131         if (currentSolutionAdvice == null) {
132             // There isn't a solution seen yet... so we are at the pipeline level
133             // Add the Precondition if it isn't here
134             pipelineAdvice.addPreconditionAdvice(advice);
135         } else {
136             // There is a current solution...
137             // Check to make sure this wasn't saved on the pipeline first
138             if (!pipelineAdvice.preconditionExists(advice)) {
139 
140                 if (advice.getTransformedRequest() != null) {
141                     currentSolutionAdvice.addRequestPreconditionAdvice(advice);
142                 }
143 
144                 if (advice.getTransformedResponse() != null) {
145                     currentSolutionAdvice.addResponsePreconditionAdvice(advice);
146                 }
147             }
148         }
149 
150     }

152     @Override
153     public void logIt() {
154         computeTransactionTimes();
155         log.info(Markers2.append("transactionData", this), "{}", getMessage());
156         logPipelineDetail();
157         logSolutionDetail();
158     }
159 
160     private void logPipelineDetail() {
161     	if (pipelineAdvice == null) {
162     		return;
163     	}
164         XformLogDetail logDetail = createLogDetail("pipeline", pipelineAdvice.getName());
165 
166         log.info(Markers2.append(XFORM_TRANSACTION_DATA, logDetail), "{}", "");
167     }
168 
169     private void logSolutionDetail() {
170     	if (pipelineAdvice == null) {
171     		return;
172     	}
173         pipelineAdvice.getSolutionAdviceList().forEach(s -> {
174             XformSolutionLogDetail logDetail = createSolutionLogDetail("solution", s.getName());
175 
176             logDetail.setProcessError(s.isProcessError());
177             logDetail.setPipelineName(pipelineAdvice.getName());
178 
179             if (!s.getRequestOperationAdviceList().isEmpty()) {
180                 logDetail.setDirection("request");
181                 log.info(Markers2.append(XFORM_TRANSACTION_DATA, logDetail), "{}", "");
182             }
183 
184             if (!s.getResponseOperationAdviceList().isEmpty()) {
185                 logDetail.setDirection("response");
186                 log.info(Markers2.append(XFORM_TRANSACTION_DATA, logDetail), "{}", "");
187             }
188         });
189     }

191     private XformLogDetail createLogDetail(String concept, String name) {
192         XformLogDetail logDetail = new XformLogDetail();
193         return initializeLogDetail(logDetail, concept, name);
194     }
195 
196     private XformSolutionLogDetail createSolutionLogDetail(String concept, String name) {
197         XformSolutionLogDetail logDetail = new XformSolutionLogDetail();
198         initializeLogDetail(logDetail, concept, name);
199         return logDetail;
200     }
201 
202     private XformLogDetail initializeLogDetail(XformLogDetail logDetail, String concept, String name) {
203         logDetail.setEventId(this.getEventId());
204         logDetail.setConcept(concept);
205         logDetail.setDestinationId(this.getDestination().getId());
206         logDetail.setOrganizationId(pipelineAdvice.getOrganizationId());
207         logDetail.setName(name);
208         logDetail.setProcessError(this.getHasProcessError());
209         logDetail.setRequestMessageType(this.getRequestPayloadType().toString());
210         logDetail.setRequestSendingApplication(this.getRequestMsh3());
211         logDetail.setRequestSendingFacility(this.getRequestMsh4());
212         logDetail.setRequestReceivingApplication(this.getRequestMsh5());
213         logDetail.setRequestReceivingFacility(this.getRequestMsh6());
214         logDetail.setResponseSendingApplication(this.getResponseMsh3());
215         logDetail.setResponseSendingFacility(this.getResponseMsh4());
216         logDetail.setResponseReceivingApplication(this.getResponseMsh5());
217         logDetail.setResponseReceivingFacility(this.getResponseMsh6());
218 
219         return logDetail;
220     }
