11  /**
12   * Validator for the {@link ValidOrganization} annotation that checks that an organization identifier exists and is active.
13   */
14  @Component
15  public class OrganizationValidator implements ConstraintValidator<ValidOrganization, UUID> {
16  
17      private final OrganizationValidationService validationService;
18  
19      /**
20       * Creates the validator.
21       * @param validationService the service that validates organizations
22       */
23      @Autowired
24      public OrganizationValidator(OrganizationValidationService validationService) {
25          this.validationService = validationService;
26      }
27  
28      @Override
29      public boolean isValid(UUID organizationId, ConstraintValidatorContext context) {
30          return validationService.validateOrganization(organizationId, context);
31      }
32  }
