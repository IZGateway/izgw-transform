1   package gov.cdc.izgateway.xform.endpoints.hub;
2   
3   import org.apache.commons.lang3.StringUtils;
4   
5   import gov.cdc.izgateway.model.RetryStrategy;
6   import gov.cdc.izgateway.soap.fault.Fault;
7   import gov.cdc.izgateway.soap.fault.MessageSupport;
8   
9   /**
10   * Represents faults related to the Transformation controller.
11   *
12   * @author Audacious Inquiry
13   */
14  public class HubControllerFault extends Fault {
15      private static final long serialVersionUID = 1L;
16  
17      private HubControllerFault(String detail, Throwable ex) {
18      	// For now, at least while in Pilot, a this Fault should indicate a CONTACT_SUPPORT RetryStrategy.
19      	super(new MessageSupport("HubControllerFault", "500", getMessage(detail), getDetail(detail), null, RetryStrategy.CONTACT_SUPPORT), ex);
20      }
21      /**
22       * Construct a new HubControllerFault with a message.
23       * @param detail the fault message; text before the first colon is the summary and text after it is the detail
24       */
25      public HubControllerFault(String detail) {
26          this(detail, null);
27      }
28      /**
29       * Construct a new HubControllerFault from an exception.
30       * @param ex the exception that caused the fault
31       */
32      public HubControllerFault(Throwable ex) {
33          this(ex.getMessage(), ex);
34      }
35      
36      private static String getMessage(String message) {
37      	return StringUtils.defaultIfEmpty(StringUtils.substringBefore(message, ":"), "An error occurred while processing the request");
38      }
39      
40      private static String getDetail(String message) {
41      	return StringUtils.defaultIfEmpty(StringUtils.substringAfter(message, ":"), message);
42      }
43  
44  }
