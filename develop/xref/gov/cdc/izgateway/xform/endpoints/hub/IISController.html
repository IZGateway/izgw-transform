31  /**
32   * SOAP endpoint for {@code /IISService} that transforms messages and sends them to the IIS destination.
33   */
34  @RestController
35  @RolesAllowed({Roles.XFORM_SENDING_SYSTEM, Roles.ADMIN})
36  @RequestMapping("/IISService")
37  @Lazy(false)
38  @Slf4j
39  public class IISController extends BaseController {
40      /**
41       * Creates the controller and registers it with the access control registry.
42       * @param mshService the service that processes MSH segments
43       * @param registry the access control registry
44       * @param producerTemplate the template used to send messages to Camel routes
45       * @param organizationService the service used to find the sending organization
46       * @param accessControlService the access control service
47       */
48      public IISController(
49              IMessageHeaderService mshService,
50              AccessControlRegistry registry,
51              ProducerTemplate producerTemplate,
52              OrganizationService organizationService,
53              AccessControlService accessControlService
54      ) {
55          super(mshService, SoapMessage.IIS2014_NS, "cdc-iis.wsdl", Arrays.asList(SoapMessage.IIS2014_NS), producerTemplate, organizationService, accessControlService);
56          registry.register(this);
57      }
58  
59      @Override
60      protected ResponseEntity<?> submitSingleMessage(SubmitSingleMessageRequest submitSingleMessage, String destinationId) throws Fault {
61  
62          return super.submitSingleMessage(submitSingleMessage, EndpointUris.DIRECT_IIS_PIPELINE);
63      }
64  
65      @Override
66      protected boolean isHubWsdl() {
67          // IIS Controller is never to be considered Hub
68          return false;
69      }
70  
71      @Operation(
72              summary = "Post a message to the SOAP Interface",
73              description = "Send a request to the SOAP Interface for IZ Gateway"
74      )
75      @ApiResponse(
76              responseCode = "200",
77              description = "The request completed normally",
78              content = {
79                      @Content(mediaType = "application/xml")
80              }
81      )
82      @ApiResponse(
83              responseCode = "500",
84              description = "A fault occured while processing the request",
85              content = {@Content}
86      )
87      @PostMapping(
88              produces = {"application/soap+xml", "application/soap", "application/xml", "text/xml", "text/plain", "text/html"}
89      )
90      @Override
91      public ResponseEntity<?> submitSoapRequest(@RequestBody SoapMessage soapMessage, @Schema(description = "Throws the fault specified in the header parameter") @RequestHeader(value = "X-IIS-Hub-Dev-Action",required = false) String devAction) throws SecurityFault {
92          return super.submitSoapRequest(soapMessage, devAction);
93      }
94  
95      protected ServiceContext createServiceContext(UUID organization, SubmitSingleMessageRequest submitSingleMessage) throws Fault {
96          try {
97              return new ServiceContext(organization,
98                      EndpointUris.IZGTS_IISService,
99                      EndpointUris.IIS_IISService,
100                     DataType.HL7V2,
101                     submitSingleMessage.getFacilityID(),
102                     submitSingleMessage.getHl7Message());
103         } catch (HL7Exception e) {
104             throw new HubControllerFault(e);
105         }
106     }

108 }
