75              IMessageHeaderService mshService,

80      ) {
81          // The base schema for HUB messages is still the iis-2014 schema, with the exception of HubHeader and certain faults.
82          super(mshService, SoapMessage.IIS2014_NS, "cdc-iis.wsdl", Arrays.asList(SoapMessage.IIS2014_NS), producerTemplate, organizationService, accessControlService);
83          this.producerTemplate = producerTemplate;

88      @Override
89      protected ResponseEntity<?> submitSingleMessage(SubmitSingleMessageRequest submitSingleMessage, String destinationId) throws Fault {
90          RequestContext.getDestinationInfo().setId(destinationId);
91          UUID organization = getOrganization(RequestContext.getSourceInfo().getCommonName()).getId();
92  
93          if (isLoopback()) {
94          	return loopbackSingleMessage(submitSingleMessage);
95          }
96          	
97          IZGXformContext context = createXformContext(organization, submitSingleMessage);
98  
99      	String transformedRequest = null;
100     	SubmitSingleMessageResponse response = null;
101         try {
102     		producerTemplate.sendBody(EndpointUris.DIRECT_HUB_PIPELINE, context);
103         	PipelineAdvice advice = XformAdviceCollector.getTransactionData().getPipelineAdvice();
104     		response = context.getSubmitSingleMessageResponse(); 
105         	
106             if (advice != null) {
107                 if (advice.isRequestTransformed()) {
108                 	transformedRequest = advice.getTransformedRequest();
109                 	if (response != null) {
110                 		response.getXformHeader().setTransformedRequest(transformedRequest);
111                 	}
112                 }
113                 if (response != null && advice.isResponseTransformed()) {
114                 	response.getXformHeader().setOriginalResponse(advice.getResponse());
115                 }
116             }
117             
118         	response.setHl7Message(context.getServiceContext().getResponseMessage().encode());
119             response.updateAction(isHubWsdl());
120         } catch (CamelExecutionException | HL7Exception e) {
121             throw new HubControllerFault(e.getCause());
122         }
123         
124         return checkResponseEntitySize(new ResponseEntity<>(response, HttpStatus.OK));
125     }

127     /**
128      * @return true if this message is requesting loopback processing, false otherwise.
129      */
130     private boolean isLoopback() {
131         List<String> headers = RequestContext.getHttpHeaders().get("x-loopback");
132         return headers != null && headers.stream().anyMatch(v -> "true".equalsIgnoreCase(v));
133 	}

135 	private ResponseEntity<?> loopbackSingleMessage(SubmitSingleMessageRequest submitSingleMessage) throws Fault  {
136         UUID organization = getOrganization(RequestContext.getSourceInfo().getCommonName()).getId();
137     	IZGXformContext context = createXformContext(organization, submitSingleMessage);
138 
139     	String transformedRequest = null;
140     	boolean isResponse = submitSingleMessage.getHl7Message().contains("MSA|");
141     	SubmitSingleMessageResponse response = new SubmitSingleMessageResponse(submitSingleMessage, getMessageNamespace(), true);
142     	try {
143         	if (isResponse) {
144         		// Setup for the Response Path
145             	response.setHl7Message(submitSingleMessage.getHl7Message());
146         		context.setSubmitSingleMessageResponse(response);
147         		context.getServiceContext().setCurrentDirection(DataFlowDirection.RESPONSE);
148         		context.getServiceContext().setResponseMessage(Hl7Utils.parseHl7v2Message(response.getHl7Message()));
149         	}
150         	
151     		producerTemplate.sendBody(EndpointUris.LOOPBACK_HUB_PIPELINE, context);
152         	PipelineAdvice advice = XformAdviceCollector.getTransactionData().getPipelineAdvice();
153         	
154             if (advice != null) {
155                 if (advice.isRequestTransformed()) {
156                 	transformedRequest = advice.getTransformedRequest();
157             		response.getXformHeader().setTransformedRequest(transformedRequest);
158                 }
159                 if (advice.isResponseTransformed()) {
160                 	response.getXformHeader().setOriginalResponse(advice.getResponse());
161                 }
162             }
163             if (isResponse) {
164             	response.setHl7Message(context.getServiceContext().getResponseMessage().encode());
165             } else {
166             	response.setHl7Message(transformedRequest);
167             }
168 
169             response.updateAction(isHubWsdl());
170         } catch (CamelExecutionException e) {
171             throw new HubControllerFault(e.getCause());
172         } catch (HL7Exception e) {
173 			throw new HubControllerFault(e.getCause());
174 		}
175         
176         return checkResponseEntitySize(new ResponseEntity<>(response, HttpStatus.OK));
177 	}

179 	protected ServiceContext createServiceContext(UUID organization, SubmitSingleMessageRequest submitSingleMessage) throws Fault {
180         try {
181             return new ServiceContext(organization,
182                     EndpointUris.IZGTS_IISHubService,
183                     EndpointUris.IZGHUB_IISHubService,
184                     DataType.HL7V2,
185                     submitSingleMessage.getFacilityID(),
186                     submitSingleMessage.getHl7Message());
187         } catch (HL7Exception e) {
188             throw new HubControllerFault(e);
189         }
190     }
