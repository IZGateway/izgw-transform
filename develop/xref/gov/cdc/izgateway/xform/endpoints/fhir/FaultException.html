1   package gov.cdc.izgateway.xform.endpoints.fhir;
2   
3   import org.apache.commons.lang3.StringUtils;
4   import org.hl7.fhir.r4.model.Extension;
5   import org.hl7.fhir.r4.model.StringType;
6   import org.hl7.fhir.r4.model.OperationOutcome.OperationOutcomeIssueComponent;
7   
8   import com.ainq.fhir.utils.PathUtils;
9   
10  import gov.cdc.izgateway.model.RetryStrategy;
11  import gov.cdc.izgateway.soap.message.FaultMessage;
12  import lombok.Getter;
13  
14  /**
15   * This class creates an exception from a FaultMessage and
16   * provided access to the FaultMessage content.
17   */
18  class FaultException extends Exception {
19      private static final long serialVersionUID = 1L;
20      @Getter
21      private final FaultMessage fault;
22      /**
23       * Create a new FaultException from a FaultMessage
24       * @param fault    The FaultMessage
25       */
26      FaultException(FaultMessage fault) {
27          super(getMessage(fault));
28          this.fault = fault;
29      }
30      
31      private static String getMessage(FaultMessage fault) {
32          if (StringUtils.isBlank(fault.getDetail())) {
33              return fault.getSummary();
34          }
35          return fault.getSummary() + ": " + fault.getDetail();
36      }
37  
38      RetryStrategy getRetryCoding() {
39          try {
40              return RetryStrategy.valueOf(fault.getRetry());
41          } catch (Exception ex) {
42              // Do nothing, we don't recognize the retry code.
43              // Preferable to suppress this "error" than the one that is being reported by this code
44              return null;
45          }
46      }
47      
48      void setOriginalText(OperationOutcomeIssueComponent issue) {
49          String original = fault.getOriginal();
50          if (StringUtils.isNotBlank(original)) {
51              issue.getDetails()
52                  .addExtension(
53                      new Extension(
54                          PathUtils.FHIR_EXT_PREFIX + "originalText",
55                          new StringType(original)
56                      )
57                  );
58          }
59      }
60          
61      void setEventId(OperationOutcomeIssueComponent issue) {
62          // Give them the eventId from the fault
63          if (StringUtils.isNotEmpty(fault.getEventId())) {
64              issue.addExtension(
65                  new Extension(
66                      PathUtils.FHIR_EXT_PREFIX + "operationoutcome-message-id",
67                      new StringType(fault.getEventId())
68                  )
69              );
70          }
71      }        
72  }
