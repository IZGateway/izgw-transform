1   package gov.cdc.izgateway.xform.api;
2   
3   import com.fasterxml.jackson.core.JsonProcessingException;
4   import gov.cdc.izgateway.security.AccessControlRegistry;
5   import gov.cdc.izgateway.xform.model.User;
6   import gov.cdc.izgateway.xform.security.Roles;
7   import gov.cdc.izgateway.xform.services.UserService;
8   import jakarta.annotation.security.RolesAllowed;
9   import jakarta.validation.Valid;
10  import lombok.extern.java.Log;
11  import org.springframework.beans.factory.annotation.Autowired;
12  import org.springframework.context.annotation.Lazy;
13  import org.springframework.http.HttpStatus;
14  import org.springframework.http.ResponseEntity;
15  import org.springframework.web.bind.annotation.*;
16  
17  import java.util.UUID;
18  import java.util.logging.Level;
19  
20  /**
21   * REST API for the users.
22   */
23  @Log
24  @Lazy(false)
25  @RestController
26  public class UserApiController extends BaseApiController {
27  
28      private final UserService userService;
29  
30      /**
31       * Creates the controller and registers it with the access control registry.
32       *
33       * @param userService the service for the users
34       * @param registry the access control registry
35       */
36      @Autowired
37      public UserApiController(
38              UserService userService,
39              AccessControlRegistry registry
40      ) {
41          super();
42          this.userService = userService;
43          registry.register(this);
44      }
45  
46      /**
47       * Gets one page of users.
48       *
49       * @param nextCursor the id of the user after which the page starts, or null
50       * @param prevCursor the id of the user before which the page ends, or null
51       * @param includeInactive if true, inactive users are included
52       * @param limit the maximum number of users in the page
53       * @return the page as JSON, or status 500 if the page cannot be serialized
54       */
55      @RolesAllowed({Roles.ADMIN})
56      @GetMapping("/api/v1/users")
57      public ResponseEntity<String> getUsersList(
58              @RequestParam(required = false) String nextCursor,
59              @RequestParam(required = false) String prevCursor,
60              @RequestParam(defaultValue = "false") Boolean includeInactive,
61              @RequestParam(defaultValue = "10") int limit
62      ) {
63          try {
64              return processList(userService.getList(), nextCursor, prevCursor, includeInactive, limit);
65          } catch (JsonProcessingException e) {
66              log.log(Level.SEVERE, e.getMessage(), e);
67              return new ResponseEntity<>(e.getMessage(), HttpStatus.INTERNAL_SERVER_ERROR);
68          }
69      }
70  
71      /**
72       * Gets one user.
73       *
74       * @param uuid the id of the user
75       * @return the user, or status 404 if it is not found
76       */
77      @RolesAllowed({Roles.ADMIN})
78      @GetMapping("/api/v1/users/{uuid}")
79      public ResponseEntity<User> getUserByUUID(@PathVariable UUID uuid) {
80          User entity = userService.getObject(uuid);
81          if (entity == null) {
82              return new ResponseEntity<>(HttpStatus.NOT_FOUND);
83          }
84          return new ResponseEntity<>(entity, HttpStatus.OK);
85      }
86  
87      /**
88       * Replaces a user.
89       *
90       * @param uuid the id of the user
91       * @param updatedUser the new user
92       * @return the updated user
93       */
94      @RolesAllowed({Roles.ADMIN})
95      @PutMapping("/api/v1/users/{uuid}")
96      public ResponseEntity<User> updateUser(
97              @PathVariable UUID uuid,
98              @RequestBody @Valid User updatedUser
99      ) {
100         updatedUser.setId(uuid);
101         userService.update(updatedUser);
102         return new ResponseEntity<>(updatedUser, HttpStatus.OK);
103     }
104 
105     /**
106      * Creates a user with a new id.
107      *
108      * @param user the user to create
109      * @return the created user
110      */
111     @RolesAllowed({Roles.ADMIN})
112     @PostMapping("/api/v1/users")
113     public ResponseEntity<User> createUser(
114             @Valid @RequestBody() User user
115     ) {
116         userService.create(user);
117         return new ResponseEntity<>(user, HttpStatus.OK);
118     }
119 
120 }
