1   package gov.cdc.izgateway.xform.api;
2   
3   import com.fasterxml.jackson.core.JsonProcessingException;
4   import gov.cdc.izgateway.security.AccessControlRegistry;
5   import gov.cdc.izgateway.xform.model.AccessControl;
6   import gov.cdc.izgateway.xform.security.Roles;
7   import gov.cdc.izgateway.xform.services.AccessControlService;
8   import jakarta.annotation.security.RolesAllowed;
9   import jakarta.validation.Valid;
10  import lombok.extern.java.Log;
11  import org.springframework.beans.factory.annotation.Autowired;
12  import org.springframework.beans.factory.annotation.Value;
13  import org.springframework.context.annotation.Lazy;
14  import org.springframework.http.HttpStatus;
15  import org.springframework.http.ResponseEntity;
16  import org.springframework.web.bind.annotation.*;
17  
18  import java.util.UUID;
19  import java.util.logging.Level;
20  
21  /**
22   * REST API for managing access control entries.
23   */
24  @Log
25  @Lazy(false)
26  @RestController
27  public class AccessControlApiController extends BaseApiController {
28      private final AccessControlService accessControlService;
29  
30      /**
31       * Create the access control API controller.
32       *
33       * @param accessControlService Service for access control operations
34       * @param registry Registry to register this controller for access control
35       */
36      @Autowired
37      public AccessControlApiController(
38              AccessControlService accessControlService,
39              AccessControlRegistry registry
40      ) {
41          this.accessControlService = accessControlService;
42          registry.register(this);
43      }
44  
45      /**
46       * List access control entries with optional pagination.
47       *
48       * @param nextCursor Cursor for the next page
49       * @param prevCursor Cursor for the previous page
50       * @param includeInactive Whether to include inactive entries
51       * @param limit Maximum number of entries to return
52       * @return A serialized page of access control entries
53       */
54      @RolesAllowed({Roles.ADMIN})
55      @GetMapping("/api/v1/access-controls")
56      public ResponseEntity<String> getAccessControlList(
57              @RequestParam(required = false) String nextCursor,
58              @RequestParam(required = false) String prevCursor,
59              @RequestParam(defaultValue = "false") Boolean includeInactive,
60              @RequestParam(defaultValue = "10") int limit
61      ) {
62          try {
63              return processList(accessControlService.getList(), nextCursor, prevCursor, includeInactive, limit);
64          } catch (JsonProcessingException e) {
65              log.log(Level.SEVERE, e.getMessage(), e);
66              return new ResponseEntity<>(e.getMessage(), HttpStatus.INTERNAL_SERVER_ERROR);
67          }
68      }
69  
70      /**
71       * Fetch a specific access control entry by id.
72       *
73       * @param uuid Access control id
74       * @return The access control entry if found
75       */
76      @RolesAllowed({Roles.ADMIN})
77      @GetMapping("/api/v1/access-controls/{uuid}")
78      public ResponseEntity<AccessControl> getAccessControlByUUID(@PathVariable UUID uuid) {
79          AccessControl entity = accessControlService.getObject(uuid);
80          if (entity == null) {
81              return new ResponseEntity<>(HttpStatus.NOT_FOUND);
82          }
83          return new ResponseEntity<>(entity, HttpStatus.OK);
84      }
85  
86      /**
87       * Update an access control entry.
88       *
89       * @param uuid Access control id
90       * @param updatedAccessControl Updated access control data
91       * @return The updated access control entry
92       */
93      @RolesAllowed({Roles.ADMIN})
94      @PutMapping("/api/v1/access-controls/{uuid}")
95      public ResponseEntity<AccessControl> updateAccessControl(
96              @PathVariable UUID uuid,
97              @RequestBody @Valid AccessControl updatedAccessControl
98      ) {
99          updatedAccessControl.setId(uuid);
100         accessControlService.update(updatedAccessControl);
101         return new ResponseEntity<>(updatedAccessControl, HttpStatus.OK);
102     }
103 
104     /**
105      * Create a new access control entry.
106      *
107      * @param accessControl New access control entry
108      * @return The created access control entry
109      */
110     @RolesAllowed({Roles.ADMIN})
111     @PostMapping("/api/v1/access-controls")
112     public ResponseEntity<AccessControl> createAccessControl(
113             @Valid @RequestBody() AccessControl accessControl
114     ) {
115         accessControlService.create(accessControl);
116         return new ResponseEntity<>(accessControl, HttpStatus.OK);
117     }
118 
119     /**
120      * Delete an access control entry.
121      *
122      * @param uuid Access control id
123      * @return HTTP 204 if deleted
124      */
125     @RolesAllowed({Roles.ADMIN})
126     @DeleteMapping("/api/v1/access-controls/{uuid}")
127     public ResponseEntity<AccessControl> deleteAccessControl(
128             @PathVariable UUID uuid
129     ) {
130         accessControlService.delete(uuid);
131         return new ResponseEntity<>(HttpStatus.NO_CONTENT);
132     }
133 }
