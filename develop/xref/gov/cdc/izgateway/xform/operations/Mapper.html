1   package gov.cdc.izgateway.xform.operations;
2   
3   import gov.cdc.izgateway.xform.context.ServiceContext;
4   import gov.cdc.izgateway.xform.enums.DataType;
5   import gov.cdc.izgateway.xform.exceptions.OperationException;
6   import jakarta.validation.constraints.NotBlank;
7   import lombok.Getter;
8   import lombok.Setter;
9   
10  /**
11   * Maps a code and code system in a message to a different code with the organization's mappings.
12   */
13  @Getter
14  @Setter
15  public class Mapper extends BaseOperation implements Operation {
16  
17      @NotBlank(message = "required and cannot be empty")
18      private String codeField;
19      @NotBlank(message = "required and cannot be empty")
20      private String codeSystemField;
21      @NotBlank(message = "required and cannot be empty")
22      private String codeSystemDefault;
23  
24      private static final org.slf4j.Logger log = org.slf4j.LoggerFactory.getLogger(Mapper.class);
25  
26      /**
27       * Creates an empty operation.
28       */
29      protected Mapper() {}
30  
31      /**
32       * Creates an operation with the settings of another operation.
33       * @param mapper the operation to copy the settings from
34       */
35      protected Mapper(Mapper mapper) {
36          super(mapper.getId(), mapper.getOrder());
37          this.codeField = mapper.getCodeField();
38          this.codeSystemField = mapper.getCodeSystemField();
39          this.codeSystemDefault = mapper.getCodeSystemDefault();
40      }
41  
42      @Override
43      public void execute(ServiceContext context) throws OperationException {
44          if (log.isTraceEnabled()) {
45              log.trace("MAP Operation: {} / CODE FIELD {} CODE SYSTEM FIELD {}", this.getClass().getSimpleName(), this.getCodeField(), this.getCodeSystemField());
46          }
47  
48          if (context.getDataType().equals(DataType.HL7V2)) {
49              Hl7v2Mapper copy = new Hl7v2Mapper(this);
50              copy.execute(context);
51          }
52      }
53  }
