4   import gov.cdc.izgateway.xform.enums.DataType;

6   import jakarta.validation.constraints.NotBlank;
7   import lombok.Getter;
8   import lombok.Setter;
9   
10  /**
11   * Sets a message field to a fixed value.
12   */
13  @Getter
14  @Setter
15  public class Set extends BaseOperation implements Operation {
16  
17      @NotBlank(message = "required and cannot be empty")
18      private String destinationField;
19      @NotBlank(message = "required and cannot be empty")
20      private String setValue;
21  
22      private static final org.slf4j.Logger log = org.slf4j.LoggerFactory.getLogger(Set.class);
23  
24      /**
25       * Creates an empty operation.
26       */
27      public Set() {}
28  
29      /**
30       * Creates an operation with the settings of another operation.
31       * @param set the operation to copy the settings from
32       */
33      protected Set(Set set) {
34          super(set.getId(), set.getOrder());
35          this.destinationField = set.destinationField;
36          this.setValue = set.setValue;
37      }
38  
39  
40      @Override
41      public void execute(ServiceContext context) throws OperationException {
42          if (log.isTraceEnabled()) {
43              log.trace("SET Operation: {} / SET {} TO {}", this.getClass().getSimpleName(), this.destinationField, this.setValue);
44          }
45  
46          if (context.getDataType().equals(DataType.HL7V2)) {
47              Hl7v2Set hl7Set = new Hl7v2Set(this);
48              hl7Set.execute(context);
49          }
50      }
51  }
