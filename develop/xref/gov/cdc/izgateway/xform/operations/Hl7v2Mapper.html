8   import gov.cdc.izgateway.xform.exceptions.OperationException;
9   import gov.cdc.izgateway.xform.model.Code;

13  import lombok.extern.slf4j.Slf4j;
14  import org.apache.commons.lang3.StringUtils;
15  
16  
17  /**
18   * HL7 v2 implementation of the {@link Mapper} operation.
19   */
20  @Slf4j
21  public class Hl7v2Mapper extends Mapper implements Operation {
22  
23      private final MappingService mappingService;
24      /**
25       * Creates an HL7 v2 mapper operation with the settings of another operation.
26       * @param mapper the operation to copy the settings from
27       */
28      public Hl7v2Mapper(Mapper mapper) {
29          super(mapper);
30          this.mappingService = SpringContext.getBean(MappingService.class);
31      }
32  
33      @Override
34      @CaptureXformAdvice
35      public void execute(ServiceContext context) throws OperationException {
36  
37          try {
38  
39              Message message = context.getCurrentMessage();
40              Terser terser = new Terser(message);
41              Code codeToMap = getCode(terser);
42              Mapping mapping = mappingService.getMapping(context.getOrganizationId(), codeToMap);
43  
44              if (mapping == null) {
45                  log.debug(String.format("Mapping not found for %s / %s", codeToMap.codeSystem(), codeToMap.code()));
46                  return;
47              }
48  
49              terser.set(this.getCodeField(), mapping.getTargetCode());
50              terser.set(this.getCodeSystemField(), mapping.getTargetCodeSystem());
51              context.setCurrentMessage(message);
52          } catch (HL7Exception ex) {
53              throw new OperationException(ex.getMessage(), ex.getCause());
54          }
55      }
56  
57      private Code getCode(Terser terser) throws HL7Exception {
58          String code = terser.get(this.getCodeField());
59          String codeSystem = StringUtils.isEmpty(terser.get(this.getCodeSystemField())) ?
60                  this.getCodeSystemDefault() : terser.get(this.getCodeSystemField());
61  
62          return new Code(code, codeSystem);
63      }
64  }
