23  /**
24   * The LogController provides access to in memory logging data on a server.
25   * This is used for integration testing to verify log content is as expected
26   * when sending messages.
27   */
28  
29  @RestController
30  @CrossOrigin
31  // TODO: Presently, blacklisted users are allowed to access the logs request, b/c blacklisting only
32  // applies to the SOAP Stack.  Once we apply it to the full HTTP stack, we will have to provide
33  // SECURE mechanism for clearing the blacklisted state of the testing user.  It cannot be said
34  // to have been applied to the full stack until this loophole is resolved.
35  @RolesAllowed({ Roles.ADMIN, Roles.OPERATIONS, Roles.BLACKLIST })
36  @RequestMapping({"/rest"})
37  @Lazy(false)
38  public class LogController extends LogControllerBase {
39  
40      /**
41       * Creates the controller and registers it for access control.
42       *
43       * @param registry the access control registry
44       */
45      @Autowired
46      public LogController(AccessControlRegistry registry) {
47          registry.register(this);
48      }
49  
50      @Operation(summary = "Get the most recent log records",
51              description = "Search for the log records matching the search parameter or all records if there is no search value")
52      @ApiResponse(responseCode = "200", description = "Success",
53              content = {
54                      @Content(mediaType = "application/json", array = @ArraySchema(schema = @Schema(implementation = LogEvent.class)))
55              }
56      )
57      @GetMapping("/logs")
58      @Override
59      public List<LogEvent> getLogs(
60              @Parameter(description = "The search string")
61              @RequestParam(required = false) String search) {
62          return super.getLogs(search);
63      }
64  
65      @Operation(summary = "Clear log records")
66      @ApiResponse(responseCode = "204", description = "Reset the logs", content = @Content)
67      @DeleteMapping("/logs")
68      @ResponseStatus(HttpStatus.NO_CONTENT)
69      @RolesAllowed({ Roles.ADMIN, Roles.OPERATIONS, Roles.BLACKLIST })
70      @Override
71      public void deleteLogs(HttpServletRequest servletReq,
72                             @Parameter(description="If true, reset the specified endpoint, clearing maintenance")
73                             @RequestParam(required = false) String clear) throws SecurityFault {
74          super.deleteLogs(servletReq, clear);
75      }
76  }
