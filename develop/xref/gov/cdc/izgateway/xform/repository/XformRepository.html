1   package gov.cdc.izgateway.xform.repository;
2   
3   import java.util.Set;
4   import java.util.UUID;
5   
6   /**
7    * Simple CRUD interface for the storage of configuration entities.
8    *
9    * @param <T> the entity type
10   */
11  public interface XformRepository<T> {
12      /**
13       * Gets an entity by its identifier.
14       *
15       * @param id the entity identifier
16       * @return the entity, or null if it does not exist
17       */
18      public T getEntity(UUID id);
19      /**
20       * Gets all entities in the repository.
21       *
22       * @return the set of entities
23       */
24      public Set<T> getEntitySet();
25      /**
26       * Adds an entity to the repository.
27       *
28       * @param obj the entity to add
29       */
30      public void createEntity(T obj);
31      /**
32       * Replaces the stored entity that has the same identifier.
33       *
34       * @param obj the entity to update
35       */
36      public void updateEntity(T obj);
37      /**
38       * Deletes an entity by its identifier.
39       *
40       * @param id the entity identifier
41       */
42      public void deleteEntity(UUID id);
43  }
