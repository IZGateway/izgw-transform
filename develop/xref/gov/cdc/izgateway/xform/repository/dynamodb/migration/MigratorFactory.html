3   import gov.cdc.izgateway.configuration.DynamoDbConfig;
4   import gov.cdc.izgateway.xform.model.BaseModel;
5   import lombok.extern.slf4j.Slf4j;
6   import org.springframework.boot.autoconfigure.condition.ConditionalOnExpression;
7   import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
8   import org.springframework.context.annotation.Bean;
9   import org.springframework.context.annotation.Configuration;
10  import software.amazon.awssdk.enhanced.dynamodb.DynamoDbEnhancedClient;
11  
12  import java.util.List;
13  import java.util.stream.Collectors;
14  
15  /**
16   * Factory that creates EntityMigrator instances for all configured entity types.
17   * This eliminates the need for individual migrator @Component classes.
18   */
19  @Slf4j
20  @Configuration
21  @ConditionalOnExpression("'${spring.database:}'.equalsIgnoreCase('migrate')")
22  public class MigratorFactory {
23  
24      private final DynamoDbEnhancedClient dynamoDbClient;
25      private final String tableName;
26      private final MigrationConfiguration migrationConfig;
27  
28      /**
29       * Creates the factory.
30       *
31       * @param dynamoDbClient the DynamoDB enhanced client
32       * @param dynamoDbConfig the DynamoDB configuration that supplies the table name
33       * @param migrationConfig the configuration of the entity file paths
34       */
35      public MigratorFactory(DynamoDbEnhancedClient dynamoDbClient,
36                             DynamoDbConfig dynamoDbConfig,
37                             MigrationConfiguration migrationConfig) {
38          this.dynamoDbClient = dynamoDbClient;
39          this.tableName = dynamoDbConfig.getDynamodbTable();
40          this.migrationConfig = migrationConfig;
41      }
42  
43      /**
44       * Creates EntityMigrator instances for all configured entity types.
45       * Spring will automatically inject this list into DataMigrationService.
46       * @return the migrators for all configured entity types
47       */
48      @Bean
49      public List<EntityMigrator<?>> entityMigrators() {
50          return migrationConfig.getEntityConfigurations()
51                  .entrySet()
52                  .stream()
53                  .map(entry -> createMigrator(entry.getKey(), entry.getValue()))
54                  .collect(Collectors.toList());
55      }
56  
57      /**
58       * Creates a generic migrator for the specified entity type.
59       */
60      @SuppressWarnings("unchecked")
61      private <T extends BaseModel> EntityMigrator<T> createMigrator(Class<T> entityClass, String filePath) {
62          log.info("Creating migrator for {} with file path: {}", entityClass.getSimpleName(), filePath);
63          return new GenericEntityMigrator<>(entityClass, filePath, dynamoDbClient, tableName);
64      }
65  }
