1   package gov.cdc.izgateway.xform.repository.dynamodb.migration;
2   
3   import gov.cdc.izgateway.xform.model.BaseModel;
4   
5   /**
6    * Interface for entity-specific migration from file storage to DynamoDB.
7    *
8    * There would be an implementation of this for each entity type (Organization, Pipeline, etc...)
9    *
10   * @param <T> Entity type to migrate
11   */
12  public interface EntityMigrator<T extends BaseModel> {
13  
14      /**
15       * Performs the migration for this entity type.
16       * Each migrator handles its own logging and throws exceptions on failure.
17       *
18       * @return MigrationCounts with total, migrated, and skipped counts
19       * @throws RuntimeException if migration fails
20       */
21      MigrationCounts migrate();
22  
23      /**
24       * Gets the entity type this migrator handles.
25       *
26       * @return The Class of the entity type
27       */
28      Class<T> getEntityType();
29  
30      /**
31       * Gets a human-readable name for this entity type.
32       *
33       * @return The entity name for logging purposes
34       */
35      String getEntityName();
36  }
