7   import lombok.extern.slf4j.Slf4j;
8   import software.amazon.awssdk.enhanced.dynamodb.DynamoDbEnhancedClient;
9   import software.amazon.awssdk.enhanced.dynamodb.TableSchema;
10  
11  import java.util.LinkedHashSet;
12  import java.util.Set;
13  
14  /**
15   * Generic migrator for entities that extend from BaseModel
16   *
17   * @param <T> Entity type to migrate
18   */
19  @Slf4j
20  public class GenericEntityMigrator<T extends BaseModel> implements EntityMigrator<T> {
21  
22      private final GenericFileRepository<T> fileRepository;
23      private final GenericDynamoDBRepository<T> dynamoDbRepository;
24      private final Class<T> entityClass;
25  
26      /**
27       * Creates a migrator that copies entities from a JSON file to a DynamoDB table.
28       *
29       * @param entityClass the entity class
30       * @param filePath the path of the JSON file to read
31       * @param dynamoDbClient the DynamoDB enhanced client
32       * @param tableName the name of the DynamoDB table
33       */
34      public GenericEntityMigrator(Class<T> entityClass,
35                                   String filePath,
36                                   DynamoDbEnhancedClient dynamoDbClient,
37                                   String tableName) {
38          this.entityClass = entityClass;
39          this.fileRepository = new GenericFileRepositoryImpl<>(filePath, entityClass);
40          this.dynamoDbRepository = new GenericDynamoDBRepositoryImpl<>(dynamoDbClient, tableName, entityClass);
41      }
42  
43      @Override
44      public MigrationCounts migrate() {
45          Set<T> entities = fileRepository.getEntitySet();
46          int totalCount = entities.size();
47          int migratedCount = 0;
48          int skippedCount = 0;
49  
50          log.info("Found {} {} entities in file storage", totalCount, getEntityName());
51  
52          for (T entity : entities) {
53              try {
54                  T existing = dynamoDbRepository.getEntity(entity.getId());
55                  if (existing != null) {
56                      log.info("{} {} already exists in DynamoDB, skipping", getEntityName(), entity.getId());
57                      skippedCount++;
58                      continue;
59                  }
60  
61                  dynamoDbRepository.createEntity(entity);
62                  migratedCount++;
63                  log.info("Migrated {}: {}", getEntityName(), entity.getId());
64  
65              } catch (Exception e) {
66                  log.warn("Failed to migrate {} {}: {}",
67                          getEntityName(), entity.getId(), e.getMessage());
68                  // Keep going with other entities
69              }
70          }
71  
72          log.info("Successfully migrated {}/{} {} ({} skipped)",
73                  migratedCount, totalCount, getEntityName(), skippedCount);
74  
75          return new MigrationCounts(totalCount, migratedCount, skippedCount);
76      }
77  
78      @Override
79      public Class<T> getEntityType() {
80          return entityClass;
81      }
82  
83      @Override
84      public String getEntityName() {
85          return entityClass.getSimpleName();
86      }
87  
88      /**
89       * Generic file repository implementation for migration
90       */
91      private static class GenericFileRepositoryImpl<T extends BaseModel> extends GenericFileRepository<T> {
92          private final Class<T> entityClass;
93  
94          public GenericFileRepositoryImpl(String filePath, Class<T> entityClass) {
95              this.filePath = filePath;
96              this.entityClass = entityClass;
97          }
98  
99          @Override
100         protected TypeReference<LinkedHashSet<T>> getTypeReference() {
101             // Create a TypeReference that uses the concrete entity class
102             return new TypeReference<LinkedHashSet<T>>() {
103                 @Override
104                 public java.lang.reflect.Type getType() {
105                     return com.fasterxml.jackson.databind.type.TypeFactory.defaultInstance()
106                             .constructCollectionType(LinkedHashSet.class, entityClass);
107                 }
108             };
109         }
110     }
111 
112     /**
113      * Generic DynamoDB repository implementation for migration
114      */
115     private static class GenericDynamoDBRepositoryImpl<T extends BaseModel> extends GenericDynamoDBRepository<T> {
116         private final Class<T> entityClass;
117 
118         public GenericDynamoDBRepositoryImpl(DynamoDbEnhancedClient dynamoDbClient,
119                                              String tableName,
120                                              Class<T> entityClass) {
121             super(dynamoDbClient, tableName, entityClass, TableSchema.fromBean(entityClass));
122             this.entityClass = entityClass;
123         }
124 
125         @Override
126         protected String getEntityName() {
127             return entityClass.getSimpleName();
128         }
129     }
130 }
