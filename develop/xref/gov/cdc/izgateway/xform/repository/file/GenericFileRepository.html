1   package gov.cdc.izgateway.xform.repository.file;
2   
3   import com.fasterxml.jackson.core.type.TypeReference;
4   import com.fasterxml.jackson.databind.ObjectMapper;
5   import gov.cdc.izgateway.xform.model.BaseModel;
6   import gov.cdc.izgateway.xform.repository.RepositoryRuntimeException;
7   import gov.cdc.izgateway.xform.repository.XformRepository;
8   
9   import java.io.IOException;
10  import java.io.InputStream;
11  import java.nio.file.Files;
12  import java.nio.file.Paths;
13  import java.util.LinkedHashSet;
14  import java.util.Set;
15  import java.util.UUID;
16  
17  /**
18   * Generic file-backed repository for configuration entities.
19   *
20   * @param <T> Model type stored in the repository
21   */
22  public abstract class GenericFileRepository<T extends BaseModel> implements XformRepository<T> {
23  
24  
25      /** The entities read from the file, or null if the file has not been read. */
26      protected LinkedHashSet<T> entities;
27      /** The path of the JSON file. */
28      protected String filePath;
29  
30      /**
31       * Creates the repository.
32       */
33      public GenericFileRepository() {
34          // The file path is set by the subclass.
35      }
36  
37      /**
38       * Provide the type reference used to deserialize the stored entities.
39       *
40       * @return Type reference for a linked hash set of entities
41       */
42      protected abstract TypeReference<LinkedHashSet<T>> getTypeReference();
43  
44      @Override
45      public T getEntity(UUID id) {
46          return getEntitySet().stream().filter(p -> p.getId().equals(id)).findFirst().orElse(null);
47      }
48  
49      @Override
50      public Set<T> getEntitySet() {
51          if (entities == null) {
52              ObjectMapper mapper = new ObjectMapper();
53              try (InputStream inputStream = Files.newInputStream(Paths.get(filePath))) {
54                  entities = mapper.readValue(inputStream, getTypeReference());
55              } catch (IOException e) {
56                  throw new RepositoryRuntimeException(String.format("Error reading file: %s", e.getMessage()), e);
57              }
58          }
59          return entities;
60      }
61  
62      @Override
63      public void createEntity(T obj) {
64          getEntitySet().add(obj);
65          writeEntitiesToFile();
66      }
67  
68      @Override
69      public void updateEntity(T obj) {
70          entities.removeIf(p -> p.getId().equals(obj.getId()));
71          createEntity(obj);
72      }
73  
74      @Override
75      public void deleteEntity(UUID id) {
76          entities.removeIf(p -> p.getId().equals(id));
77          writeEntitiesToFile();
78      }
79  
80      private void writeEntitiesToFile() {
81          ObjectMapper mapper = new ObjectMapper();
82          try {
83              String json = mapper.writerWithDefaultPrettyPrinter().writeValueAsString(entities);
84              Files.write(Paths.get(filePath), json.getBytes());
85          } catch (IOException e) {
86              throw new RepositoryRuntimeException("Error writing file.", e);
87          }
88      }
89  }
