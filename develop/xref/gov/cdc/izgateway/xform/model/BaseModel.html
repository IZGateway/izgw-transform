1   package gov.cdc.izgateway.xform.model;
2   
3   import com.fasterxml.jackson.annotation.JsonIgnore;
4   import software.amazon.awssdk.enhanced.dynamodb.mapper.annotations.DynamoDbAttribute;
5   import software.amazon.awssdk.enhanced.dynamodb.mapper.annotations.DynamoDbPartitionKey;
6   import software.amazon.awssdk.enhanced.dynamodb.mapper.annotations.DynamoDbSortKey;
7   
8   import java.util.UUID;
9   
10  /**
11   * Base class for the configuration entities that are stored in files or in DynamoDB.
12   */
13  public abstract class BaseModel {
14      /**
15       * Returns the unique identifier of the entity.
16       *
17       * @return the identifier
18       */
19      public abstract UUID getId();
20      /**
21       * Sets the unique identifier of the entity.
22       *
23       * @param id the identifier
24       */
25      public abstract void setId(UUID id);
26      /**
27       * Returns true if the entity is active.
28       *
29       * @return the active flag
30       */
31      public abstract Boolean getActive();
32  
33      /** The attribute to use for entity type */
34      public static final String ENTITY_TYPE = "entityType";
35      /** The attribute to use for the sort key */
36      public static final String SORT_KEY = "sortKey";
37  
38      /**
39       * Creates the entity.
40       */
41      public BaseModel() {
42      }
43  
44      /**
45       * Returns the DynamoDB partition key, which is the simple class name of the entity.
46       *
47       * @return the entity type
48       */
49      @JsonIgnore
50      @DynamoDbPartitionKey
51      @DynamoDbAttribute(ENTITY_TYPE)
52      public String getEntityType() {
53          return getClass().getSimpleName();
54      }
55  
56      /**
57       * Does nothing, because the entity type is read-only. DynamoDB needs a setter for the attribute.
58       *
59       * @param entityType ignored
60       */
61      public void setEntityType(String entityType) {
62          // Read-only attribute
63      }
64  
65      /**
66       * Returns the DynamoDB sort key, which is the identifier as a string.
67       *
68       * @return the sort key, or null if the identifier is null
69       */
70      @JsonIgnore
71      @DynamoDbSortKey
72      @DynamoDbAttribute(SORT_KEY)
73      public String getSortKey() {
74          return getId() != null ? getId().toString() : null;
75      }
76  
77      /**
78       * Sets the identifier from the DynamoDB sort key.
79       *
80       * @param sortKey the identifier as a string, or null
81       */
82      public void setSortKey(String sortKey) {
83          setId(sortKey != null ? UUID.fromString(sortKey) : null);
84      }
85  }
