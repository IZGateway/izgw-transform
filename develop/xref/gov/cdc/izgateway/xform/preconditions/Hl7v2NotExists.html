8   public class Hl7v2NotExists extends NotExists implements Precondition {
9   
10      /**
11       * Creates an empty precondition.
12       */
13      public Hl7v2NotExists() {
14          super();
15      }
16  
17      /**
18       * Creates an HL7 v2 precondition with the settings of another precondition.
19       * @param notExists the precondition to copy the settings from
20       */
21      public Hl7v2NotExists(NotExists notExists) {
22          super(notExists);
23      }
24  
25      @Override
26      public boolean evaluate(ServiceContext context) {
27          Hl7v2Exists exists = new Hl7v2Exists(this);
28          return !exists.evaluate(context);
29      }
30  }
