1   package gov.cdc.izgateway.xform.preconditions;
2   
3   import ca.uhn.hl7v2.HL7Exception;
4   import ca.uhn.hl7v2.model.Message;
5   import ca.uhn.hl7v2.util.Terser;
6   import gov.cdc.izgateway.xform.context.ServiceContext;
7   
8   /**
9    * HL7 v2 implementation of the {@link Equals} precondition.
10   */
11  public class Hl7v2Equals extends Equals implements Precondition {
12  
13      /**
14       * Creates an empty precondition.
15       */
16      public Hl7v2Equals() {}
17  
18      /**
19       * Creates a precondition that compares a message field to a value.
20       * @param dataPath the HL7 v2 Terser path of the field
21       * @param comparisonValue the value to compare the field to
22       */
23      public Hl7v2Equals(String dataPath, String comparisonValue) {
24          super(dataPath, comparisonValue);
25      }
26  
27      /**
28       * Creates an HL7 v2 precondition with the settings of another precondition.
29       * @param equals the precondition to copy the settings from
30       */
31      public Hl7v2Equals(Equals equals) {
32          super(equals);
33      }
34  
35      @Override
36      public boolean evaluate(ServiceContext context) {
37  
38          Message message = context.getCurrentMessage();
39          // TODO: If IZGW returns a fault message, message will be null.
40          // We need to figure out how to handle that
41          if (message == null) {
42          	return false;
43          }
44          Terser terser = new Terser(message);
45  
46          String sourceValue;
47          try {
48              sourceValue = terser.get(this.getDataPath());
49          } catch (HL7Exception e) {
50              // HL7Exception will happen if Terser.get references Segment that does not exist in the message
51              // In this case just returning false.  Entirely possible that Analyst/Engineer has a precondition
52              // set for a segment that just simply won't exist in every message submitted.
53              return false;
54          }
55  
56          if (sourceValue == null) {
57              return false;
58          }
59  
60          return (sourceValue.equals(this.getComparisonValue()));
61  
62      }
63  }
