1   package gov.cdc.izgateway.xform.preconditions;
2   
3   import ca.uhn.hl7v2.HL7Exception;
4   import ca.uhn.hl7v2.model.Message;
5   import ca.uhn.hl7v2.util.Terser;
6   import gov.cdc.izgateway.xform.context.ServiceContext;
7   
8   import java.util.regex.Matcher;
9   
10  /**
11   * HL7 v2 implementation of the {@link RegexMatch} precondition.
12   */
13  public class Hl7v2RegexMatch extends RegexMatch implements Precondition {
14  
15      /**
16       * Creates an HL7 v2 precondition with the settings of another precondition.
17       * @param regexMatch the precondition to copy the settings from
18       */
19      public Hl7v2RegexMatch(RegexMatch regexMatch) {
20          super(regexMatch);
21      }
22  
23      @Override
24      public boolean evaluate(ServiceContext context) {
25  
26          Message message = context.getCurrentMessage();
27          if (message == null) {
28          	// If there was a fault, message will be null.
29          	return false;
30          }
31          Terser terser = new Terser(message);
32  
33          String sourceValue;
34          try {
35              sourceValue = terser.get(this.getDataPath());
36          } catch (HL7Exception e) {
37              // HL7Exception will happen if Terser.get references Segment that does not exist in the message
38              // In this case just returning false.  Entirely possible that Analyst/Engineer has a precondition
39              // set for a segment that just simply won't exist in every message submitted.
40              return false;
41          }
42  
43          if (sourceValue == null) {
44              return false;
45          }
46  
47          Matcher matcher = getMatcher(sourceValue);
48          return matcher.matches();
49      }
50  
51  }
