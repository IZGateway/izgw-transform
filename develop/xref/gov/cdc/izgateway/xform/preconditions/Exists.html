1   package gov.cdc.izgateway.xform.preconditions;
2   
3   import gov.cdc.izgateway.xform.annotations.CaptureXformAdvice;
4   import gov.cdc.izgateway.xform.context.ServiceContext;
5   import gov.cdc.izgateway.xform.enums.DataType;
6   import gov.cdc.izgateway.xform.logging.advice.Advisable;
7   import gov.cdc.izgateway.xform.logging.advice.Transformable;
8   import jakarta.validation.constraints.NotNull;
9   import lombok.Getter;
10  import lombok.Setter;
11  import lombok.extern.slf4j.Slf4j;
12  
13  import java.util.UUID;
14  
15  /**
16   * A precondition that is true if a value exists at a data path.
17   */
18  @Slf4j
19  @Getter
20  @Setter
21  public class Exists implements Precondition, Advisable, Transformable {
22      @NotNull(message = "required and cannot be empty")
23      private UUID id;
24      @NotNull(message = "required and cannot be empty")
25      private String dataPath;
26  
27      /**
28       * Creates an empty precondition.
29       */
30      protected Exists() {
31      }
32  
33      /**
34       * Creates a precondition with the same id and data path as another one.
35       *
36       * @param exists the precondition to copy
37       */
38      protected Exists(Exists exists) {
39          this.id = exists.id;
40          this.dataPath = exists.getDataPath();
41      }
42  
43      /**
44       * Creates a precondition for a data path.
45       *
46       * @param dataPath the path of the value to check
47       */
48      protected Exists(String dataPath) {
49          this.dataPath = dataPath;
50      }
51  
52      @Override
53      @CaptureXformAdvice
54      public boolean evaluate(ServiceContext context) {
55          if (log.isTraceEnabled()) {
56              log.trace("Precondition: {} / dataPath: '{}'",
57                      this.getClass().getSimpleName(),
58                      this.getDataPath());
59          }
60  
61          if (this.dataPath.startsWith("state.")) {
62              String stateKey = this.dataPath.split("\\.")[1];
63              return context.getState().containsKey(stateKey);
64          } else if (context.getDataType().equals(DataType.HL7V2)) {
65              return new Hl7v2Exists(this).evaluate(context);
66          }
67  
68          return false;
69      }
70  
71      @Override
72      public String getName() {
73          return this.getClass().getSimpleName();
74      }
75  
76      @Override
77      public boolean hasTransformed() {
78          return true;
79      }
80  }
