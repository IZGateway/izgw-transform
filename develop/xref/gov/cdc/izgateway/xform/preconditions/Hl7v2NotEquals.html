5   /**
6    * HL7 v2 implementation of the {@link NotEquals} precondition.
7    */
8   public class Hl7v2NotEquals extends NotEquals implements Precondition {
9       /**
10       * Creates an empty precondition.
11       */
12      public Hl7v2NotEquals() {}
13  
14      /**
15       * Creates an HL7 v2 precondition with the settings of another precondition.
16       * @param notEquals the precondition to copy the settings from
17       */
18      public Hl7v2NotEquals(NotEquals notEquals) {
19          super(notEquals);
20      }
21  
22      @Override
23      public boolean evaluate(ServiceContext context) {
24          Hl7v2Equals hl7v2Equals = new Hl7v2Equals(this.getDataPath(), this.getComparisonValue());
25          return !hl7v2Equals.evaluate(context);
26      }
27  }
