3   import gov.cdc.izgateway.xform.model.OperationPreconditionField;
4   import gov.cdc.izgateway.xform.repository.RepositoryFactory;
5   import org.springframework.beans.factory.annotation.Autowired;
6   import org.springframework.stereotype.Service;
7   
8   import java.util.ArrayList;
9   import java.util.List;
10  
11  /**
12   * Service for the field definitions that operations and preconditions use.
13   */
14  @Service
15  public class OperationPreconditionFieldService extends GenericService<OperationPreconditionField> {
16      /**
17       * Creates the service.
18       *
19       * @param repositoryFactory the factory that supplies the field repository
20       */
21      @Autowired
22      protected OperationPreconditionFieldService(RepositoryFactory repositoryFactory) {
23          super(repositoryFactory.operationPreconditionFieldRepository());
24      }
25  
26      /**
27       * Gets the accessible fields that are marked for use by operations.
28       *
29       * @return the operation fields
30       */
31      public List<OperationPreconditionField> getOperationList() {
32          List<OperationPreconditionField> allFields = super.getList();
33          List<OperationPreconditionField> operationFields = new ArrayList<>();
34          for (OperationPreconditionField field : allFields) {
35              if (Boolean.TRUE.equals(field.getForOperation())) {
36                  operationFields.add(field);
37              }
38          }
39          return operationFields;
40      }
41  
42      /**
43       * Gets the accessible fields that are marked for use by preconditions.
44       *
45       * @return the precondition fields
46       */
47      public List<OperationPreconditionField> getPreconditionList() {
48          List<OperationPreconditionField> allFields = super.getList();
49          List<OperationPreconditionField> preconditionFields = new ArrayList<>();
50          for (OperationPreconditionField field : allFields) {
51              if (Boolean.TRUE.equals(field.getForPrecondition())) {
52                  preconditionFields.add(field);
53              }
54          }
55          return preconditionFields;
56      }
57  
58      @Override
59      protected boolean isDuplicate(OperationPreconditionField operationPreconditionField) {
60          return repo.getEntitySet().stream().anyMatch(
61                  opf -> opf.getDataPath().equalsIgnoreCase(operationPreconditionField.getDataPath()) &&
62                          opf.getFieldName().equalsIgnoreCase(operationPreconditionField.getFieldName())
63          );
64      }
65  
66  
67  }
