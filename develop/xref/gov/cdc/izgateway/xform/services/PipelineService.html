1   package gov.cdc.izgateway.xform.services;

3   import gov.cdc.izgateway.xform.model.Pipeline;
4   import gov.cdc.izgateway.xform.repository.RepositoryFactory;
5   
6   import org.apache.commons.lang3.Strings;
7   import org.springframework.beans.factory.annotation.Autowired;
8   import org.springframework.stereotype.Service;
9   import java.util.UUID;
10  
11  /**
12   * Service for the pipelines.
13   */
14  @Service
15  public class PipelineService extends GenericService<Pipeline>{
16  
17      /**
18       * Creates the service.
19       *
20       * @param repositoryFactory the factory that supplies the pipeline repository
21       */
22      @Autowired
23      public PipelineService(RepositoryFactory repositoryFactory) {
24          super(repositoryFactory.pipelineRepository());
25      }
26  
27      /**
28       * Finds the active pipeline for an organization and endpoint pair.
29       *
30       * @param organizationId the id of the organization
31       * @param inboundEndpoint the inbound endpoint
32       * @param outboundEndpoint the outbound endpoint
33       * @return the pipeline, or null if no active pipeline matches
34       */
35      public Pipeline getPipelineByOrganizationAndEndpoints(UUID organizationId, String inboundEndpoint, String outboundEndpoint) {
36          return getList()
37                  .stream()
38                  .filter(
39                          p -> p.getOrganizationId().equals(organizationId)
40                                  && p.getInboundEndpoint().equals(inboundEndpoint)
41                                  && p.getOutboundEndpoint().equals(outboundEndpoint)
42                                  && Boolean.TRUE.equals(p.getActive())
43                  ).findFirst().orElse(null);
44      }
45  
46      @Override
47      protected boolean isDuplicate(Pipeline pipeline) {
48          return isPipelineNameDuplicate(pipeline) || isEndpointDuplicate(pipeline);
49      }
50  
51      /**
52       * Checks if the organization of a pipeline already has a pipeline with the same name, ignoring case.
53       *
54       * @param pipeline the pipeline to check
55       * @return true if the name is already in use
56       */
57      protected boolean isPipelineNameDuplicate(Pipeline pipeline) {
58          return repo.getEntitySet().stream().anyMatch(p ->
59                  p.getOrganizationId().equals(pipeline.getOrganizationId()) &&
60                          Strings.CI.equals(p.getPipelineName(), pipeline.getPipelineName())
61          );
62      }
63  
64      /**
65       * Checks if the organization of a pipeline already has a pipeline with the same endpoints, ignoring case.
66       *
67       * @param pipeline the pipeline to check
68       * @return true if the endpoints are already in use
69       */
70      protected boolean isEndpointDuplicate(Pipeline pipeline) {
71          return repo.getEntitySet().stream().anyMatch(p ->
72                  p.getOrganizationId().equals(pipeline.getOrganizationId()) &&
73                          Strings.CI.equals(p.getInboundEndpoint(), pipeline.getInboundEndpoint()) &&
74                          Strings.CI.equals(p.getOutboundEndpoint(), pipeline.getOutboundEndpoint())
75          );
76      }
77      
78      /**
79       * Determines if a solution is in use by any pipeline
80       * @param solutionId	The solution UUID to check
81       * @return			True if the solution is in use, false otherwise
82       */
83      public boolean isSolutionInUse(UUID solutionId) {
84          return repo.getEntitySet().stream()
85                  .flatMap(p -> p.getPipes().stream())
86                  .filter(pipe -> pipe.getSolutionId().equals(solutionId))
87                  .findFirst()
88                  .orElse(null) != null;
89      }
90  
91  }
