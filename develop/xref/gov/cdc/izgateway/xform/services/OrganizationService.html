1   package gov.cdc.izgateway.xform.services;
2   
3   import gov.cdc.izgateway.xform.model.Organization;
4   import gov.cdc.izgateway.xform.repository.RepositoryFactory;
5   import org.springframework.beans.factory.annotation.Autowired;
6   import org.springframework.stereotype.Service;
7   
8   /**
9    * Service for the organizations.
10   */
11  @Service
12  public class OrganizationService  extends GenericService<Organization> {
13      /**
14       * Creates the service.
15       *
16       * @param repositoryFactory the factory that supplies the organization repository
17       */
18      @Autowired
19      public OrganizationService(RepositoryFactory repositoryFactory) {
20          super(repositoryFactory.organizationRepository());
21      }
22  
23      /**
24       * Finds the active organization with a common name.
25       *
26       * @param commonName the common name
27       * @return the organization, or null if no active organization has the common name
28       */
29      public Organization getOrganizationByCommonName(String commonName) {
30          return repo.getEntitySet().stream().filter(o -> o.getCommonName().equals(commonName) && Boolean.TRUE.equals(o.getActive())).findFirst().orElse(null);
31      }
32  
33      /**
34       * Checks if an active organization with a common name exists.
35       *
36       * @param commonName the common name
37       * @return true if an active organization has the common name
38       */
39      public boolean organizationExists(String commonName) {
40          return repo.getEntitySet().stream().anyMatch(o -> o.getCommonName().equals(commonName) && Boolean.TRUE.equals(o.getActive()));
41      }
42  
43      /**
44       * Checks if an organization with the same organizationName and commonName already exists.
45       *
46       * @param organization The organization to check for duplication
47       * @return true if a duplicate exists, false otherwise
48       */
49      @Override
50      protected boolean isDuplicate(Organization organization) {
51          return repo.getEntitySet().stream()
52                  .anyMatch(o ->
53                          o.getOrganizationName().equalsIgnoreCase(organization.getOrganizationName()) &&
54                                  o.getCommonName().equalsIgnoreCase(organization.getCommonName())
55                  );
56      }
57  }
