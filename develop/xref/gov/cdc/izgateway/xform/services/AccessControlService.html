1   package gov.cdc.izgateway.xform.services;
2   
3   import gov.cdc.izgateway.logging.RequestContext;
4   import gov.cdc.izgateway.security.AccessControlRegistry;
5   import gov.cdc.izgateway.xform.model.AccessControl;
6   import gov.cdc.izgateway.xform.repository.RepositoryFactory;
7   import gov.cdc.izgateway.xform.security.Roles;
8   import lombok.extern.slf4j.Slf4j;
9   import org.springframework.beans.factory.annotation.Autowired;
10  import org.springframework.stereotype.Service;
11  import org.springframework.web.bind.annotation.RequestMethod;
12  import java.util.*;
13  
14  /**
15   * Service for access control operations and authorization checks.
16   */
17  @Slf4j
18  @Service
19  public class AccessControlService extends GenericService<AccessControl> {
20      private final AccessControlRegistry registry;
21  
22      /**
23       * Create the access control service.
24       *
25       * @param repositoryFactory Repository factory for access control data
26       * @param registry Access control registry
27       */
28      @Autowired
29      public AccessControlService(RepositoryFactory repositoryFactory, AccessControlRegistry registry) {
30          super(repositoryFactory.accessControlRepository());
31          this.registry = registry;
32      }
33  
34      /**
35       * Get the roles allowed for the given method and path.
36       *
37       * @param method HTTP method
38       * @param path Request path
39       * @return List of allowed roles
40       */
41      public List<String> getAllowedRoles(RequestMethod method, String path) {
42          List<String> roles = registry.getAllowedRoles(method, path);
43          log.debug("Roles allowed for {} {} are {}", method, path, roles);
44          return roles;
45      }
46  
47      /**
48       * Check whether access is allowed for the given method and path.
49       *
50       * @param method HTTP method
51       * @param path Request path
52       * @return True if access is allowed
53       */
54      public Boolean checkAccess(String method, String path) {
55          // If path starts with /swagger, return checkSwaggerAccess, else return checkXformAccess
56          if (path.startsWith("/swagger")) {
57              return checkSwaggerAccess(method, path);
58          } else {
59              return checkXformAccess(method, path);
60          }
61      }
62  
63      /**
64       * Check if the user has access to the swagger endpoints.  This method is necessary because the swagger
65       * library and endpoints will not register with our AccessControlRegistry.
66       * @param method HTTP method
67       * @param path Request path
68       * @return true if access is allowed, false if access is denied
69       */
70      public Boolean checkSwaggerAccess(String method, String path) {
71          return path.startsWith("/swagger") && RequestContext.getPrincipal().getRoles().contains(Roles.ADMIN);
72      }
73  
74      /**
75       * Check if the user has access to the xform endpoints.
76       * @param method HTTP method
77       * @param path Request path
78       * @return true if access is allowed, false if access is denied
79       */
80      public Boolean checkXformAccess(String method, String path) {
81          List<String> allowedRoles = getAllowedRoles(RequestMethod.valueOf(method), path);
82  
83          // Check for public access
84          if (allowedRoles.contains(Roles.PUBLIC_ACCESS)) {
85              return true;
86          }
87  
88          // If RequestContext.getRoles() has one role that matches the roles list, return true
89          return RequestContext.getPrincipal().getRoles().stream().anyMatch(allowedRoles::contains);
90      }
91  
92      /**
93       * Build a mapping of user ids to their assigned roles.
94       *
95       * @return Map of user id to roles
96       */
97      public Map<String, TreeSet<String>> getUserRoles() {
98          Map<String, TreeSet<String>> userRoleMap = new HashMap<>();
99          for (AccessControl ac : repo.getEntitySet()) {
100             userRoleMap.put(ac.getUserId().toString(), new TreeSet<>(ac.getRoles()));
101         }
102 
103         return userRoleMap;
104     }
105 
106     /**
107      * Checks if an access control with the same user id already exists
108      *
109      * @param accessControl The Access Control to check for duplication
110      * @return true if a duplicate exists, false otherwise
111      */
112     @Override
113     protected boolean isDuplicate(AccessControl accessControl) {
114         return repo.getEntitySet().stream()
115                 .anyMatch(ac ->
116                         ac.getUserId().equals(accessControl.getUserId())
117                 );
118     }
119 }
