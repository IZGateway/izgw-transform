1   package gov.cdc.izgateway.xform.services;

3   import gov.cdc.izgateway.security.principal.GroupToRoleMapper;
4   import gov.cdc.izgateway.xform.model.GroupRoleMapping;
5   import gov.cdc.izgateway.xform.repository.RepositoryFactory;
6   import lombok.extern.slf4j.Slf4j;
7   import org.springframework.beans.factory.annotation.Autowired;
8   import org.springframework.stereotype.Service;
9   
10  import java.util.*;
11  
12  /**
13   * Service for the mappings from user groups to roles.
14   */
15  @Slf4j
16  @Service
17  public class GroupRoleMappingService extends GenericService<GroupRoleMapping> implements GroupToRoleMapper {
18  
19      /**
20       * Creates the service.
21       *
22       * @param repositoryFactory the factory that supplies the group to role mapping repository
23       */
24      @Autowired
25      public GroupRoleMappingService(RepositoryFactory repositoryFactory) {
26          super(repositoryFactory.groupRoleMappingRepository());
27      }
28  
29      /**
30       * Gets the roles of the first mapping for a group.
31       *
32       * @param group the group name
33       * @return the roles, or an empty list if no mapping exists for the group
34       */
35      public List<String> getRolesByGroup(String group) {
36          return repo.getEntitySet().stream()
37                  .filter(m -> m.getGroupName().equals(group))
38                  .findFirst()
39                  .map(GroupRoleMapping::getRoles)
40                  .orElse(Collections.emptyList());
41      }
42  
43      /**
44       * For the given groups, it returns the roles that are mapped to the groups.
45       * @param groups The groups to map to roles
46       * @return The roles that are mapped from the groups
47       */
48      @Override
49      public Set<String> mapGroupsToRoles(Set<String> groups) {
50          Set<String> roles = new HashSet<>();
51          for (String group : groups) {
52              roles.addAll(getRolesByGroup(group));
53          }
54          return roles;
55      }
56  
57      @Override
58      protected boolean isDuplicate(GroupRoleMapping groupRoleMapping) {
59          return repo.getEntitySet().stream()
60                  .anyMatch(grp ->
61                          grp.getGroupName().equalsIgnoreCase(groupRoleMapping.getGroupName())
62                  );
63      }
64  
65  }
