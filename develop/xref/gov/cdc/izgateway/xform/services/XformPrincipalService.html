21  /**
22   * Gets the principal of a request from a JWT or a client certificate, and adds the roles and organizations of the user.
23   */
24  @Service
25  @Slf4j
26  public class XformPrincipalService implements PrincipalService {
27  
28      private final CertificatePrincipalProvider certificatePrincipalProvider;
29      private final JwtPrincipalProvider jwtPrincipalProvider;
30      private final AccessControlService accessControlService;
31      private final UserService userService;
32      private final List<String> localHostIps = Arrays.asList(HostInfo.LOCALHOST_IP4, "0:0:0:0:0:0:0:1", HostInfo.LOCALHOST_IP6);
33      private final GroupRoleMappingService groupRoleMappingService;
34  
35      /**
36       * Creates the service.
37       *
38       * @param accessControlService the service that supplies the roles of the users
39       * @param certificatePrincipalProvider the provider that creates a principal from a client certificate
40       * @param jwtPrincipalProvider the provider that creates a principal from a JWT
41       * @param userService the service for the users
42       * @param groupRoleMappingService the service for the group to role mappings
43       */
44      @Autowired
45      public XformPrincipalService(AccessControlService accessControlService,
46                                   CertificatePrincipalProvider certificatePrincipalProvider,
47                                   JwtPrincipalProvider jwtPrincipalProvider,
48                                   UserService userService,
49                                   GroupRoleMappingService groupRoleMappingService) {
50          this.certificatePrincipalProvider = certificatePrincipalProvider;
51          this.jwtPrincipalProvider = jwtPrincipalProvider;
52          this.accessControlService = accessControlService;
53          this.userService = userService;
54          this.groupRoleMappingService = groupRoleMappingService;
55      }
56  
57      /**
58       * Gets the principal from the request. This will first try to get the principal from a JWT,
59       * if that fails, it will get the principal from the client certificate,
60       * if that fails, it will return an UnauthenticatedPrincipal.
61       * @param request The request to get the principal from
62       * @return The principal from the request.  If no principal is found, an UnauthenticatedPrincipal is returned.
63       */
64      @Override
65      public IzgPrincipal getPrincipal(HttpServletRequest request) {
66          IzgPrincipal izgPrincipal = jwtPrincipalProvider.createPrincipalFromJwt(request);
67  
68          if (izgPrincipal == null) {
69              izgPrincipal = getPrincipalFromCertificate(request);
70          }
71  
72          if (izgPrincipal == null) {
73              izgPrincipal = new UnauthenticatedPrincipal();
74              // Add admin role for localhost requests, but only for the UnauthenticatedPrincipal
75              addRolesIfFromLocalhost(izgPrincipal, request);
76          }
77  
78          if ( izgPrincipal instanceof UnauthenticatedPrincipal ) {
79              return izgPrincipal;
80          } else {
81              XformPrincipal xformPrincipal = new XformPrincipal(izgPrincipal);
82              User user = userService.getUserByUserName(xformPrincipal.getName());
83              xformPrincipal.setAllowedOrganizationIds(user.getOrganizationIds());
84  
85              // Remove admin role if X-Remove-Admin-Role header is present (used for testing)
86              if (!StringUtils.isEmpty(request.getHeader(Roles.REMOVE_ADMIN_ROLE_HEADER))) {
87                  xformPrincipal.getRoles().remove(Roles.ADMIN);
88              }
89  
90              return xformPrincipal;
91          }
92      }
93  
94      /**
95       * Gets the principal from the client certificate in the request.
96       * @param request The request to get the principal from
97       * @return The principal from the client certificate in the request, null if no principal found
98       */
99      private IzgPrincipal getPrincipalFromCertificate(HttpServletRequest request) {
100         IzgPrincipal izgPrincipal = certificatePrincipalProvider.createPrincipalFromCertificate(request);
101 
102         if (izgPrincipal == null) {
103             log.warn("No principal found via certificate in the request.");
104             return null;
105         }
106 
107         User user = userService.getUserByUserName(izgPrincipal.getName());
108         if (user == null) {
109             log.warn("User not found for common name: {} - no roles will be assigned based on certificate", izgPrincipal.getName());
110             return izgPrincipal;
111         }
112 
113         // Add roles for the user as specified in the access control configuration
114         Map<String, TreeSet<String>> usersAndRoles = accessControlService.getUserRoles();
115         TreeSet<String> roles = usersAndRoles.get(user.getId().toString());
116         if (roles == null || roles.isEmpty()) {
117             log.warn("No roles found for user: {}", user.getUserName());
118             return izgPrincipal;
119         }
120 
121         izgPrincipal.setRoles(roles);
122         log.debug("Added roles {} for user: {}", roles, user.getUserName());
123 
124         return izgPrincipal;
125     }
126 
127     private void addRolesIfFromLocalhost(IzgPrincipal izgPrincipal, HttpServletRequest request) {
128         if (isLocalHost(request.getRemoteHost()) && StringUtils.isEmpty(request.getHeader(Roles.NOT_ADMIN_HEADER))) {
129             log.debug("Adding ADMIN role for localhost request");
130             izgPrincipal.getRoles().add(Roles.ADMIN);
131         }
132     }
133 
134     private boolean isLocalHost(String remoteHost) {
135         return localHostIps.contains(remoteHost);
136     }
