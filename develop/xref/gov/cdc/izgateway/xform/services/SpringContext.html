1   package gov.cdc.izgateway.xform.services;
2   
3   import org.springframework.context.ApplicationContext;
4   import org.springframework.context.ApplicationContextAware;
5   import org.springframework.context.annotation.Lazy;
6   import org.springframework.stereotype.Component;
7   
8   /**
9    * Gives static access to Spring beans for objects that Spring does not create.
10   */
11  @Component
12  @Lazy(false)
13  public class SpringContext implements ApplicationContextAware {
14  
15      private static ApplicationContext context;
16  
17      /**
18       * Creates the component. Spring sets the application context after creation.
19       */
20      public SpringContext() {
21          // No state to initialize.
22      }
23  
24      @Override
25      public void setApplicationContext(ApplicationContext applicationContext) {
26          context = applicationContext;
27      }
28  
29      /**
30       * Gets the bean of a type from the application context.
31       *
32       * @param <T> the bean type
33       * @param beanClass the bean class
34       * @return the bean
35       */
36      public static <T> T getBean(Class<T> beanClass) {
37          return context.getBean(beanClass);
38      }
39  }
