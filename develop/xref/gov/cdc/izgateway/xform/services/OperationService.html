16  /**
17   * Service that lists the operation types and their properties from the {@code @JsonSubTypes} annotation on {@link Operation}.
18   */
19  @Service
20  public class OperationService implements XformService<OperationInfo> {
21      /**
22       * Creates the service.
23       */
24      public OperationService() {
25          // No state to initialize.
26      }
27  
28      @Override
29      public OperationInfo getObject(UUID id) {
30          return null;
31      }
32  
33      @Override
34      public List<OperationInfo> getList() {
35          // Use reflection to determine the operations that exist in the running system
36          // Start with pulling the annotation on the Operation interface
37          // Then build up array of OperationInfo to show fields necessary
38          // when configuring each operation.
39          Class<?> operationClass = Operation.class;
40          JsonSubTypes subTypesAnnotation = operationClass.getAnnotation(JsonSubTypes.class);
41          List<OperationInfo> operationInfoList = new ArrayList<>();
42  
43          if (subTypesAnnotation != null) {
44              JsonSubTypes.Type[] types = subTypesAnnotation.value();
45  
46              for (JsonSubTypes.Type type : types) {
47                  Class<?> subClass = type.value();
48                  OperationInfo info = new OperationInfo();
49                  info.setMethod(type.name());
50  
51                  Map<String, OperationInfoProperty> properties = new HashMap<>();
52                  getAllFields(subClass, properties);
53                  info.setProperties(properties);
54  
55                  operationInfoList.add(info);
56              }
57          }
58          ApiEventLogger.logReadEvent(operationInfoList);
59          return operationInfoList;
60      }
61  
62      @Override
63      public void update(OperationInfo obj) {
64          // Operations are determined and listed via code reflection, not to be updated in code.
65      }
66  
67      @Override
68      public void create(OperationInfo obj) {
69          // Operations are determined and listed via code reflection, not to be created in code.
70      }
71  
72      @Override
73      public void delete(UUID id) {
74          // Operations are determined and listed via code reflection, not to be deleted in code.
75      }

77      private void getAllFields(Class<?> clazz, Map<String, OperationInfoProperty> properties) {
78          if (clazz == null || clazz == Object.class) {
79              return;
80          }
81  
82          // Process current class
83          for (Field field : clazz.getDeclaredFields()) {
84              if (shouldIncludeField(field)) {
85                  OperationInfoProperty prop = new OperationInfoProperty();
86                  prop.setType(field.getType().getSimpleName());
87                  prop.setRequired(!field.isAnnotationPresent(Nullable.class));
88                  properties.put(field.getName(), prop);
89              }
90          }
91  
92          // Process superclass
93          getAllFields(clazz.getSuperclass(), properties);
94      }
95  
96      private boolean shouldIncludeField(Field field) {
97          // If a field is annotated with @JsonIgnore in an Operation do not include this in the output.
98          // If a field is static do not include in the output.  Only non-static fields would be updated in
99          // the state so this will exclude the AspectJ objects that may show up.
100         return !field.isAnnotationPresent(JsonIgnore.class) && !Modifier.isStatic(field.getModifiers());
101     }
102 }
