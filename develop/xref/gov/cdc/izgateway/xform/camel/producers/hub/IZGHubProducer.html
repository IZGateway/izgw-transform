18  /**
19   * Camel producer that sends the transformed request to the IZ Gateway Hub destination and stores the response in the context.
20   */
21  @Slf4j
22  public class IZGHubProducer extends ProducerBase {
23  
24      /**
25       * Creates the producer.
26       * @param endpoint the endpoint that owns the producer
27       */
28      public IZGHubProducer(Endpoint endpoint) {
29          super(endpoint);
30      }
31  
32      @Override
33      public void process(Exchange exchange) throws Exception {
34          HubMessageSender messageSender = getHubComponent().getMessageSender();
35  
36          IZGXformContext context = exchange.getIn().getBody(IZGXformContext.class);
37  
38          IDestination destination = createDestination(getHubComponent());
39          destination.setUsername(context.getSubmitSingleMessageRequest().getUsername());
40          destination.setPassword(context.getSubmitSingleMessageRequest().getPassword());
41          setDestinationInfoFromDestination(RequestContext.getDestinationInfo(), destination);
42  
43          try {
44              context.getSubmitSingleMessageRequest().setHl7Message(context.getServiceContext().getRequestMessage().encode());
45          } catch (HL7Exception e) {
46              throw new HubControllerFault(e);
47          }
48  
49          SubmitSingleMessageResponse response = messageSender.sendSubmitSingleMessage(destination,
50                  context.getSubmitSingleMessageRequest());
51          context.getServiceContext().setCurrentDirection(DataFlowDirection.RESPONSE);
52          context.getServiceContext().setResponseMessage(Hl7Utils.parseHl7v2Message(response.getHl7Message()));
53          context.setSubmitSingleMessageResponse(response);
54      }
55  
56      private IZGHubComponent getHubComponent() {
57          return (IZGHubComponent) getEndpoint().getComponent();
58      }
59  }
