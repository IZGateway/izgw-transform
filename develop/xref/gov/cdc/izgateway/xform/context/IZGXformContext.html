1   package gov.cdc.izgateway.xform.context;
2   
3   import gov.cdc.izgateway.soap.message.FaultMessage;
4   import gov.cdc.izgateway.soap.message.SubmitSingleMessageRequest;
5   import gov.cdc.izgateway.soap.message.SubmitSingleMessageResponse;
6   import lombok.Getter;
7   import lombok.Setter;
8   
9   /**
10   * Context object required for a pipeline that uses the Hub or IIS outbound endpoints.
11   * 
12   * @author Audacious Inquiry
13   */
14  @Getter
15  @Setter
16  public class IZGXformContext {
17      private ServiceContext serviceContext;
18      private SubmitSingleMessageRequest submitSingleMessageRequest;
19      private SubmitSingleMessageResponse submitSingleMessageResponse;
20      private FaultMessage faultMessage;
21      
22      /**
23       * Creates a context for a request that has no response yet.
24       *
25       * @param serviceContext the service context of the message
26       * @param submitSingleMessageRequest the inbound SOAP request
27       */
28      public IZGXformContext(ServiceContext serviceContext, SubmitSingleMessageRequest submitSingleMessageRequest) {
29          this.serviceContext = serviceContext;
30          this.submitSingleMessageRequest = submitSingleMessageRequest;
31          this.submitSingleMessageResponse = null;
32      }
33  
34      /**
35       * Creates a context for a request and its response.
36       *
37       * @param serviceContext the service context of the message
38       * @param submitSingleMessageRequest the inbound SOAP request
39       * @param submitSingleMessageResponse the SOAP response
40       */
41      public IZGXformContext(ServiceContext serviceContext, SubmitSingleMessageRequest submitSingleMessageRequest, SubmitSingleMessageResponse submitSingleMessageResponse) {
42          this.serviceContext = serviceContext;
43          this.submitSingleMessageRequest = submitSingleMessageRequest;
44          this.submitSingleMessageResponse = submitSingleMessageResponse;
45      }
46      
47      /**
48       * Get the fault status.
49       * @return true if the context reflects that a fault occured.
50       */
51      public boolean hasFault() {
52      	return faultMessage != null;
53      }
54      
55      /**
56       * See if there is a response (if not, there should be a fault).
57       * @return	true if there is a response.
58       */
59      public boolean hasResponse() {
60      	return submitSingleMessageResponse != null;
61      }
62  }
