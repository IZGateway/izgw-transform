3   import gov.cdc.izgateway.xform.context.ServiceContext;
4   import gov.cdc.izgateway.xform.enums.DataFlowDirection;
5   import gov.cdc.izgateway.xform.exceptions.OperationException;
6   import gov.cdc.izgateway.xform.exceptions.SolutionOperationException;
7   import gov.cdc.izgateway.xform.operations.Operation;
8   import gov.cdc.izgateway.xform.preconditions.Precondition;
9   import lombok.extern.slf4j.Slf4j;
10  
11  import java.util.Collections;
12  import java.util.List;
13  import java.util.Objects;
14  
15  /**
16   * Runtime form of a solution operation: a list of operations that run only when all preconditions pass.
17   */
18  @Slf4j
19  public class SolutionOperation {
20      private final List<Precondition> preconditions;
21      private final List<Operation> operations;
22      private final String solutionName;
23      private final DataFlowDirection direction;
24  
25      /**
26       * Creates a runtime solution operation from its configuration and sorts its operations by order.
27       * @param solutionOperation the solution operation configuration
28       * @param solutionName the name of the solution, used in log messages
29       * @param direction the direction of the messages that this solution operation transforms
30       */
31      public SolutionOperation(gov.cdc.izgateway.xform.model.SolutionOperation solutionOperation, String solutionName, DataFlowDirection direction) {
32          this.solutionName = solutionName;
33          this.direction = direction;
34          preconditions = Objects.requireNonNullElse(solutionOperation.getPreconditions(), Collections.emptyList());
35          operations = Objects.requireNonNullElse(solutionOperation.getOperationList(), Collections.emptyList());
36          // Ensure that SolutionOperation list is sorted by order
37          Collections.sort(operations, this::compareByOrder);
38      }
39  
40  	private int compareByOrder(Operation o1, Operation o2) {
41  		if (o1 == o2) {
42  			return 0;
43  		}
44  		if (o2 == null) {
45  			return 1;
46  		}
47  		if (o1 == null) {
48  			return -1;
49  		}
50  		int order1 = Objects.requireNonNullElse(o1.getOrder(), Integer.valueOf(0));
51  		int order2 = Objects.requireNonNullElse(o2.getOrder(), Integer.valueOf(0));
52  		return Integer.compare(order1, order2);
53  	}
54  
55      private boolean passedPreconditions(ServiceContext context) {
56          boolean passed = true;
57  
58          for (Precondition op : preconditions) {
59              passed = passed && op.evaluate(context);
60          }
61  
62          return passed;
63      }
64  
65      /**
66       * Runs the operations in order if all preconditions pass.
67       * @param context the context of the message to transform
68       * @throws SolutionOperationException if an operation fails
69       */
70      public void execute(ServiceContext context) throws SolutionOperationException {
71  
72          if (passedPreconditions(context)) {
73              log.debug("Solution-level precondition passed for {} ({})", solutionName, direction);
74              for (Operation op : operations) {
75                  try {
76                      op.execute(context);
77                  } catch (OperationException e) {
78                      throw new SolutionOperationException(
79                              String.format("Failed to execute operation: %s - %s",
80                                      op.getClass().getSimpleName(),
81                                      e.getMessage()),
82                              e.getCause());
83                  }
84              }
85          } else {
86              log.debug("Solution-level precondition failed for {} ({})", solutionName, direction);
87          }
88      }
89  }
