47      /**
48       * Generate a a NIST SP 800-90A compliant secure random number
49       * generator.
50       *
51       * @return A compliant generator.
52       */
53      @Bean
54      public static SecureRandom getSecureRandom() {
55          /*
56           * According to NIST Special Publication 800-90A, a Nonce is
57           * A time-varying value that has at most a negligible chance of
58           * repeating, e.g., a random value that is generated anew for each
59           * use, a timestamp, a sequence number, or some combination of
60           * these.
61           *
62           * The nonce is combined with the entropy input to create the initial
63           * DRBG seed.
64           */
65          if (secureRandom != null) {
66              return secureRandom;
67          }
68          byte[] nonce = ByteBuffer.allocate(8).putLong(System.nanoTime()).array();
69          EntropySourceProvider entSource = new BasicEntropySourceProvider(new SecureRandom(), true);
70          FipsDRBG.Builder drgbBldr = FipsDRBG.SHA512
71                  .fromEntropySource(entSource).setSecurityStrength(256)
72                  .setEntropyBitsRequired(256);
73          secureRandom = drgbBldr.build(nonce, true);
74          return secureRandom;
75      }
