17              value = {
18                      "/MSH-3-1~MSH.3.1",
19                      "/MSH-4-1~^([A-Z]{3})\\.([0-9]{1,})\\.([0-9]{1,})$",
20                      "/PID-13-1~\\((\\d{3})\\) (\\d{3})-(\\d{4})",
21                      "/PID-13-4~(\\w+)@(\\w+)\\.(com|org|net)?",
22                      "/PID-13-4~^[A-Za-z0-9+_.-]+@(.+)$",
23                      "/PID-6-1~\\w+(?:,\\w+)*",
24                      "/PID-6-2~^(?!.*[0-9]).*$",
25                      "/PID-19-1~^\\d{3}-?\\d{2}-?\\d{4}$",
26                      "/PID-13-9~^([01]?\\d\\d?|2[0-4]\\d|25[0-5])\\.([01]?\\d\\d?|2[0-4]\\d|25[0-5])\\.([01]?\\d\\d?|2[0-4]\\d|25[0-5])\\.([01]?\\d\\d?|2[0-4]\\d|25[0-5])$"
27              })
28      void testTrueRegexes(String dataPath, String regex) throws HL7Exception {
29          ServiceContext context = new ServiceContext(UUID.randomUUID(),"","", DataType.HL7V2, "", TestMessage1());
30  
31          RegexMatch regexMatch = new RegexMatch(UUID.randomUUID(), dataPath, regex);
32  
33          assertTrue(
34                  regexMatch.evaluate(context)
35          );
36      }

50              })
51      void testStateRegexes(String valueToTest, String regex) throws HL7Exception {
52          ServiceContext context = new ServiceContext(UUID.randomUUID(),"","", DataType.HL7V2, "", TestMessage1());
53          context.getState().put("CONTEXT_KEY", valueToTest);
54          RegexMatch regexMatch = new RegexMatch(UUID.randomUUID(), "state.CONTEXT_KEY", regex);
55  
56          assertTrue(
57                  regexMatch.evaluate(context)
58          );
59      }
