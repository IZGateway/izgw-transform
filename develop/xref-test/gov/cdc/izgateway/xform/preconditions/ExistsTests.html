17      @ParameterizedTest

19              "/MSH-4-1",

25      void testHl7True(String dataPath) throws HL7Exception {
26          ServiceContext context = new ServiceContext(UUID.randomUUID(),"","", DataType.HL7V2, "", TestMessage1());
27  
28          Hl7v2Exists exists = new Hl7v2Exists();
29          exists.setDataPath(dataPath);
30  
31          assertTrue(
32                  exists.evaluate(context)
33          );
34      }

36      @Test

38          ServiceContext context = new ServiceContext(UUID.randomUUID(),"","", DataType.HL7V2, "", TestMessage1());
39          context.getState().put("CONTEXT_KEY", "CONTEXT_VALUE");
40  
41          Exists exists = new Exists();
