27  MSH|^~\\&|SendingSystem|SendingFacility|PCC|PCC|20240516120000||ADT^A08|MSGID12345|P|2.5.1\r
28  PID|1|1234567890|A1234567^^^HospitalA^MR||Doe^John^Jacob||19800101|M|||123 Main St^^Metropolis^IL^44130^USA||(555) 434-5543|||||1234567890\r
29  PV1|1|I|Ward123^Room456^Bed789|||||||||||||||||1234567890\r""".replace("\n","");
30  	@Test
31  	void test() throws HL7Exception, SolutionOperationException {
32  		// If these operations are executed in the right order, MSH 5 and 6 will both set to PCC
33  		Set opToExecuteFirst = getSetOperation("/MSH-5-1", "PCC");
34  		opToExecuteFirst.setOrder(0);
35  		Copy opToExecuteSecond = getCopyOperation("/MSH-5-1", "/MSH-6-1");
36  		opToExecuteSecond.setOrder(1);
37  
38  		List<Operation> misorderedList = new ArrayList<>();
39  		misorderedList.add(opToExecuteSecond);
40  		misorderedList.add(opToExecuteFirst);
41  		gov.cdc.izgateway.xform.model.SolutionOperation modelOp = new gov.cdc.izgateway.xform.model.SolutionOperation();
42  		// Make a copy of the list when setting it.
43  		List<Operation> sortedList = new ArrayList<>(misorderedList);  // First make a copy of misordered list.
44  		modelOp.setOperationList(sortedList);
45  		SolutionOperation actualOp = new SolutionOperation(modelOp, "TestSolution", DataFlowDirection.REQUEST);
46  		// Now sortedList is actually sorted.
47  		
48  		// Verify the list is no longer misordered.
49  		assertNotEquals(sortedList.get(0), misorderedList.get(0));
50  		assertNotEquals(sortedList.get(1), misorderedList.get(1));
51  
52  		ServiceContext serviceContext = new ServiceContext(UUID.randomUUID(), "", "", DataType.HL7V2, "", TEST_INPUT_HL7);
53  		actualOp.execute(serviceContext);
54  		
55  		String test = serviceContext.getCurrentMessage().encode();
56          assertEquals(TEST_EXPECTED_HL7, test);
57  	}
